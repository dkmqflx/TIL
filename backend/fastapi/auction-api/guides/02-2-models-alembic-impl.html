① Alembic 설치와 ruff 설정

Alembic은 운영 서버에서도 마이그레이션을 실행해야 하므로 --dev 없이 일반 의존성으로 넣습니다.

bash
uv add alembic
# 검증 당시: alembic 1.19.1

그리고 pyproject.toml의 ruff 설정 두 곳을 손봅니다. 이유가 둘 있습니다. 첫째, 잠시 뒤 프로젝트 안에 alembic/이라는 폴더가 생기는데, ruff는 그걸 보고 "alembic은 이 프로젝트의 자체 모듈이구나"라고 오해합니다. 그러면 from alembic import context를 외부 라이브러리가 아닌 내 코드 취급해서 import 정렬을 틀리게 잡습니다. known-third-party로 바로잡습니다. 둘째, Alembic이 자동 생성하는 마이그레이션 파일은 우리 스타일 규칙(줄 길이, 따옴표)을 안 지키는데, 자동 생성물을 매번 손으로 고치는 건 의미가 없으니 검사에서 뺍니다.

toml(수정) pyproject.toml — [tool.ruff] 부터 [tool.ruff.lint.isort] 까지를 이렇게 바꾼다
[tool.ruff]
target-version = "py313"
line-length = 100
extend-exclude = ["alembic/versions"]  # 자동 생성 파일은 검사 제외

[tool.ruff.lint]
select = ["E", "F", "I", "UP", "B"]

[tool.ruff.lint.isort]
known-third-party = ["alembic"]  # 프로젝트에 alembic/ 폴더가 있어 1st-party 로 오인하는 것 방지
무엇이 바뀌었나. 1단계의 [tool.ruff]에 extend-exclude 한 줄이 추가됐고, [tool.ruff.lint.isort] 섹션이 새로 생겼습니다. [tool.pytest.ini_options]는 그대로 둡니다.

② app/models/base.py — 모든 모델의 부모와 공통 컬럼

Base는 Week 5·02의 DeclarativeBase 그대로입니다. 여기에 믹스인 두 개를 추가합니다. 네 테이블 모두 id(UUID)와 created_at을 갖는데, 매번 같은 두 줄을 복사하지 않고 작은 클래스로 빼서 상속시키는 방식입니다. sort_order는 믹스인 컬럼이 테이블에서 몇 번째에 올지 정하는 옵션인데, 이게 없으면 믹스인의 컬럼은 자식 클래스 컬럼 뒤에 붙어서 id가 맨 마지막 컬럼이 되는 어색한 테이블이 됩니다.

pythonapp/models/base.py
import uuid
from datetime import datetime

from sqlalchemy import DateTime, func
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column


class Base(DeclarativeBase):
    """모든 모델의 부모. Base.metadata 에 테이블 정의가 모인다 (Alembic 이 이걸 본다)."""


class UUIDPrimaryKeyMixin:
    id: Mapped[uuid.UUID] = mapped_column(primary_key=True, default=uuid.uuid4, sort_order=-100)


class CreatedAtMixin:
    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True), server_default=func.now(), nullable=False, sort_order=100
    )

두 가지 선택을 설명하면, UUID를 기본키로 쓴 이유는 경매 ID가 URL에 노출되기 때문입니다. 정수 ID는 /auctions/1, /auctions/2처럼 다음 번호가 뻔해서 남의 경매를 순서대로 긁어갈 수 있습니다. default=uuid.uuid4는 파이썬 쪽에서 값을 만들어 INSERT하라는 뜻입니다. server_default=func.now()는 반대로 DB가 값을 채우게 하는 것인데, 여러 서버의 시계가 조금씩 달라도 DB 시계 하나를 기준으로 통일되는 장점이 있습니다.

③ app/models/enums.py — 상태와 분류

경매 상태·카테고리·결제 상태처럼 "정해진 값 중 하나"는 파이썬 StrEnum으로 정의합니다. 문자열을 상속하기 때문에 AuctionStatus.ACTIVE == "active"가 참이고, JSON으로 내보낼 때도 그냥 문자열이 됩니다. 맨 아래 enum_values는 SQLAlchemy가 DB에 저장할 값 목록을 만들어 주는 작은 도우미입니다 — 이게 없으면 SQLAlchemy는 이름("ACTIVE")을 저장하는데, 우리는 값("active")을 저장하고 싶습니다.

pythonapp/models/enums.py
from enum import StrEnum


class AuctionStatus(StrEnum):
    SCHEDULED = "scheduled"  # 시작 전
    ACTIVE = "active"  # 입찰 받는 중
    ENDED = "ended"  # 마감 (낙찰 또는 유찰)
    CANCELLED = "cancelled"  # 판매자가 취소 (입찰 0건일 때만)


class AuctionCategory(StrEnum):
    ELECTRONICS = "electronics"
    FASHION = "fashion"
    HOME = "home"
    BOOKS = "books"
    HOBBY = "hobby"
    ETC = "etc"


class PaymentStatus(StrEnum):
    PENDING = "pending"  # 낙찰 직후, 결제 대기
    PAID = "paid"
    EXPIRED = "expired"  # 기한 내 미결제


def enum_values(enum_cls: type[StrEnum]) -> list[str]:
    """Enum(values_callable=...) 용 — DB 에 이름("ACTIVE") 대신 값("active") 을 저장한다."""
    return [m.value for m in enum_cls]

⑤ app/models/auction.py — 경매 (핵심 테이블)

가장 긴 모델이니 위에서부터 읽어 내려가겠습니다. __table_args__에는 컬럼이 아닌 테이블 수준 정의(CHECK 제약, 복합 인덱스)를 넣습니다. 제약마다 name=을 꼭 붙이세요. 이름이 없으면 DB가 임의의 이름을 붙이는데, 나중에 마이그레이션으로 제약을 지우거나 바꿀 때 그 이름을 알아야 해서 곤란해집니다.

users로 가는 외래키가 둘입니다. seller_id는 판매자라 필수이고 ondelete="CASCADE"(판매자가 탈퇴하면 경매도 삭제), winner_id는 낙찰 전엔 비어 있으니 | None이고 SET NULL(낙찰자가 탈퇴해도 경매 기록은 남기되 낙찰자만 비움)입니다. 이렇게 같은 테이블로 가는 관계가 둘이면 relationship()에 foreign_keys=로 "이 관계는 이 컬럼을 쓴다"를 명시해야 합니다. 안 하면 SQLAlchemy가 어느 컬럼으로 조인할지 몰라 에러를 냅니다.

pythonapp/models/auction.py
import uuid
from datetime import datetime
from typing import TYPE_CHECKING

from sqlalchemy import (
    BigInteger,
    CheckConstraint,
    DateTime,
    Enum,
    ForeignKey,
    Index,
    String,
    Text,
    func,
)
from sqlalchemy.orm import Mapped, mapped_column, relationship

from app.models.base import Base, CreatedAtMixin, UUIDPrimaryKeyMixin
from app.models.enums import AuctionCategory, AuctionStatus, enum_values

if TYPE_CHECKING:
    from app.models.bid import Bid
    from app.models.user import User


class Auction(UUIDPrimaryKeyMixin, CreatedAtMixin, Base):
    __tablename__ = "auctions"
    __table_args__ = (
        CheckConstraint("starting_price > 0", name="ck_auctions_starting_price_positive"),
        CheckConstraint("current_price > 0", name="ck_auctions_current_price_positive"),
        CheckConstraint("min_increment > 0", name="ck_auctions_min_increment_positive"),
        CheckConstraint("ends_at > starts_at", name="ck_auctions_ends_after_starts"),
        Index("ix_auctions_status_ends_at", "status", "ends_at"),
    )

    seller_id: Mapped[uuid.UUID] = mapped_column(
        ForeignKey("users.id", ondelete="CASCADE"), index=True
    )
    title: Mapped[str] = mapped_column(String(120))
    description: Mapped[str] = mapped_column(Text, default="")
    category: Mapped[AuctionCategory] = mapped_column(
        Enum(AuctionCategory, native_enum=False, length=20, values_callable=enum_values)
    )
    starting_price: Mapped[int] = mapped_column(BigInteger)
    current_price: Mapped[int] = mapped_column(BigInteger)
    min_increment: Mapped[int] = mapped_column(BigInteger, default=1000)
    starts_at: Mapped[datetime] = mapped_column(DateTime(timezone=True))
    ends_at: Mapped[datetime] = mapped_column(DateTime(timezone=True))
    status: Mapped[AuctionStatus] = mapped_column(
        Enum(AuctionStatus, native_enum=False, length=20, values_callable=enum_values),
        default=AuctionStatus.SCHEDULED,
    )
    bid_count: Mapped[int] = mapped_column(default=0)
    winner_id: Mapped[uuid.UUID | None] = mapped_column(ForeignKey("users.id", ondelete="SET NULL"))
    version: Mapped[int] = mapped_column(default=0)
    updated_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True), server_default=func.now(), onupdate=func.now(), sort_order=101
    )

    seller: Mapped["User"] = relationship(foreign_keys=[seller_id])
    winner: Mapped["User | None"] = relationship(foreign_keys=[winner_id])
    bids: Mapped[list["Bid"]] = relationship(
        back_populates="auction", cascade="all, delete-orphan", order_by="Bid.amount.desc()"
    )

몇 가지 더 짚습니다. if TYPE_CHECKING: 블록 안의 import는 타입 검사기만 보고 실행 시에는 무시됩니다. auction.py가 bid.py를 import하고 bid.py가 다시 auction.py를 import하면 순환 참조로 터지는데, 이 방법으로 피합니다. 그래서 relationship의 타입은 따옴표로 감싼 문자열("Bid")입니다. version은 6단계 낙관적 락에서, bid_count·current_price는 6단계 입찰에서 갱신합니다. 지금은 자리만 잡아 둡니다.

⑥ app/models/bid.py — 입찰

"이 경매의 최고 입찰"을 찾는 쿼리는 WHERE auction_id = ? ORDER BY amount DESC LIMIT 1입니다. 이 쿼리에 딱 맞는 인덱스가 (auction_id, amount DESC)인데, 내림차순 인덱스는 __table_args__ 안의 문자열 방식으로는 표현할 수 없어서 클래스 밖에서 컬럼 객체로 만듭니다. Bid.amount.desc()처럼요. 파일 어디에 쓰든 Base.metadata에 등록되므로 Alembic이 똑같이 인식합니다.

pythonapp/models/bid.py
import uuid
from typing import TYPE_CHECKING

from sqlalchemy import BigInteger, CheckConstraint, ForeignKey, Index
from sqlalchemy.orm import Mapped, mapped_column, relationship

from app.models.base import Base, CreatedAtMixin, UUIDPrimaryKeyMixin

if TYPE_CHECKING:
    from app.models.auction import Auction
    from app.models.user import User


class Bid(UUIDPrimaryKeyMixin, CreatedAtMixin, Base):
    __tablename__ = "bids"
    __table_args__ = (CheckConstraint("amount > 0", name="ck_bids_amount_positive"),)

    auction_id: Mapped[uuid.UUID] = mapped_column(ForeignKey("auctions.id", ondelete="CASCADE"))
    bidder_id: Mapped[uuid.UUID] = mapped_column(
        ForeignKey("users.id", ondelete="CASCADE"), index=True
    )
    amount: Mapped[int] = mapped_column(BigInteger)

    auction: Mapped["Auction"] = relationship(back_populates="bids")
    bidder: Mapped["User"] = relationship()


# "이 경매의 최고 입찰" 을 빨리 찾기 위한 복합 인덱스 (auction_id 오름차순, amount 내림차순)
Index("ix_bids_auction_id_amount_desc", Bid.auction_id, Bid.amount.desc())

⑦ app/models/payment.py — 결제

경매 하나에 결제는 하나뿐이므로 auction_id에 unique=True를 겁니다. 이 한 줄이 "낙찰 처리가 실수로 두 번 돌아도 결제가 두 건 생기지 않는다"를 DB 수준에서 보장해 주고, 8단계 마감 워커의 멱등성을 뒷받침합니다. external_id·paid_at은 결제 전엔 비어 있으니 | None입니다.

pythonapp/models/payment.py
import uuid
from datetime import datetime

from sqlalchemy.orm import Mapped, mapped_column, relationship

from app.models.base import Base, CreatedAtMixin, UUIDPrimaryKeyMixin
from app.models.enums import PaymentStatus, enum_values

    from app.models.auction import Auction

class Payment(UUIDPrimaryKeyMixin, CreatedAtMixin, Base):

    )
    payer_id: Mapped[uuid.UUID] = mapped_column(

    external_id: Mapped[str | None] = mapped_column(String(100))  # PG 가 발급한 결제 ID (9단계)
    expires_at: Mapped[datetime] = mapped_column(DateTime(timezone=True))

    auction: Mapped["Auction"] = relationship()

⑧ app/models/__init__.py — 한 곳에서 전부 import

⑨ 테스트부터 — tests/test_models.py

DB에 테이블을 만들기 전에 테스트를 씁니다. 세 가지를 확인합니다. 모델 메타데이터에 네 테이블이 선언돼 있는가, DB에 실제로 네 테이블이 있는가, 그리고 경매 한 건을 넣어 보면 기본값(status, min_increment 등)이 채워지고 ends_at ≤ starts_at인 데이터는 DB가 거부하는가.

아직 테스트 전용 DB와 자동 롤백 장치는 없습니다(3단계). 그래서 이 테스트는 개발 DB에 붙되 절대 커밋하지 않습니다. flush()는 INSERT를 DB에 보내되 트랜잭션은 열어 둔 상태라, 마지막에 rollback()하면 흔적이 남지 않습니다. 또 하나, pytest-asyncio가 아직 없어서 asyncio.run()으로 코루틴을 돌리는데, 이벤트 루프를 닫기 전에 engine.dispose()로 그 루프에서 만든 커넥션을 정리해야 합니다. 안 그러면 다음 테스트가 "닫힌 루프의 커넥션"을 꺼내 쓰다 에러가 납니다.

pythontests/test_models.py
"""2단계 스모크 테스트 — 모델 정의와 실제 DB 스키마가 맞물리는지 확인한다.

아직 테스트 전용 DB·롤백 fixture(3단계)가 없으므로, 개발 DB 에 연결하되
커밋은 절대 하지 않고 rollback 으로 흔적을 남기지 않는다.
"""

import asyncio
import uuid

def run(coro: Coroutine[Any, Any, Any]) -> Any:
    """코루틴을 실행하고, 이 이벤트 루프에서 만든 커넥션을 반드시 정리한다."""

    async def wrapper() -> Any:
        try:
            return await coro
        finally:
            await engine.dispose()

    return asyncio.run(wrapper())

def test_auction_defaults_and_check_constraint():
    async def scenario() -> None:
        async with SessionLocal() as session:
            seller = User(
                email=f"seller-{uuid.uuid4()}@example.com", password_hash="x", nickname="셀러"
            )
            session.add(seller)
            await session.flush()  # INSERT 는 나가지만 커밋은 아니다 → seller.id 가 채워진다

            now = datetime.now(UTC)
            auction = Auction(
                seller_id=seller.id,
                title="중고 키보드",
                category=AuctionCategory.ELECTRONICS,
                starting_price=10_000,
                current_price=10_000,
                starts_at=now,
                ends_at=now + timedelta(hours=1),
            )
            session.add(auction)
            await session.flush()
            await session.refresh(auction)  # server_default(created_at 등) 를 DB 에서 읽어 온다

            assert auction.status == AuctionStatus.SCHEDULED
            assert auction.min_increment == 1_000
            assert auction.bid_count == 0
            assert auction.version == 0
            assert auction.created_at.tzinfo is not None

            # ends_at <= starts_at 은 DB 의 CHECK 제약이 막아야 한다
            bad = Auction(
                seller_id=seller.id,
                title="잘못된 경매",
                category=AuctionCategory.ETC,
                starting_price=1_000,
                current_price=1_000,
                starts_at=now,
                ends_at=now,
            )
            session.add(bad)
            with pytest.raises(IntegrityError):
                await session.flush()

            await session.rollback()  # 지금까지의 INSERT 를 전부 되돌린다 → DB 는 깨끗

    run(scenario())

⑩ Alembic 초기화

-t async는 "비동기 드라이버(asyncpg)용 템플릿으로"라는 뜻입니다. 마지막 인자 alembic은 만들어질 폴더 이름입니다. 실행하면 alembic.ini(설정)와 alembic/ 폴더(env.py, script.py.mako, versions/)가 생깁니다.

bash
uv run alembic init -t async alembic
# Creating directory .../alembic ...  done
# Generating .../alembic/env.py ...  done
# Generating .../alembic.ini ...  done
# Please edit configuration/connection/logging settings in .../alembic.ini before proceeding.

생성된 alembic.ini에는 sqlalchemy.url = driver://user:pass@localhost/dbname이라는 자리표시 줄이 있습니다. 여기에 진짜 주소를 적으면 비밀번호가 git에 올라가고, .env와 이중 관리가 됩니다. 그래서 이 줄은 주석 처리하고, 잠시 뒤 env.py에서 1단계의 Settings로부터 주소를 주입합니다. 아래 명령이 그 줄을 찾아 주석으로 바꿉니다(직접 편집기로 열어 줄 앞에 #을 붙여도 됩니다).

bash
sed -i '' 's|^sqlalchemy.url = driver://user:pass@localhost/dbname|# sqlalchemy.url 은 env.py 에서 Settings(.env) 로 주입한다\n# sqlalchemy.url =|' alembic.ini
grep -n "sqlalchemy.url" alembic.ini
# 89:# sqlalchemy.url 은 env.py 에서 Settings(.env) 로 주입한다
# 90:# sqlalchemy.url =
Linux라면 sed -i '' 대신 sed -i(빈 따옴표 없이)입니다. macOS의 sed만 -i 뒤에 백업 확장자를 요구합니다. grep 결과의 줄 번호(89·90)는 Alembic 버전에 따라 조금 다를 수 있습니다.

⑪ alembic/env.py — 우리 프로젝트에 맞게 전면 교체

템플릿이 만들어 준 env.py를 아래 내용으로 통째로 바꿉니다. 템플릿과 다른 곳은 세 군데뿐입니다. config.set_main_option("sqlalchemy.url", ...)으로 .env의 주소를 주입하는 것, target_metadata = Base.metadata로 "현재 모델"을 알려 주는 것, 그리고 compare_type=True로 컬럼 타입 변경도 감지하게 하는 것입니다. 나머지는 Week 10·02에서 본 async 템플릿 그대로입니다 — Alembic 자체는 동기 라이브러리라서 run_sync로 감싸 실행합니다.

pythonalembic/env.py
import asyncio
from logging.config import fileConfig

from alembic import context
from sqlalchemy import pool
from sqlalchemy.engine import Connection
from sqlalchemy.ext.asyncio import async_engine_from_config

from app.core.config import get_settings
from app.models import Base

# Alembic Config 객체 — alembic.ini 의 값에 접근한다
config = context.config

# alembic.ini 의 sqlalchemy.url 대신 앱의 Settings(.env) 에서 접속 주소를 가져온다.
# → 앱과 마이그레이션이 같은 DB 를 보게 되고, 비밀번호가 ini 에 박히지 않는다.
config.set_main_option("sqlalchemy.url", get_settings().database_url)

if config.config_file_name is not None:
    fileConfig(config.config_file_name)

# autogenerate 가 "현재 모델" 로 삼는 메타데이터
target_metadata = Base.metadata


def run_migrations_offline() -> None:
    """DB 연결 없이 SQL 만 출력하는 모드 (alembic upgrade head --sql)."""
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
        compare_type=True,
    )
    with context.begin_transaction():
        context.run_migrations()


def do_run_migrations(connection: Connection) -> None:
    context.configure(connection=connection, target_metadata=target_metadata, compare_type=True)
    with context.begin_transaction():
        context.run_migrations()


async def run_async_migrations() -> None:
    """asyncpg 엔진으로 연결한 뒤, 동기 API 인 Alembic 을 run_sync 로 감싸 실행한다."""
    connectable = async_engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )
    async with connectable.connect() as connection:
        await connection.run_sync(do_run_migrations)
    await connectable.dispose()


def run_migrations_online() -> None:
    asyncio.run(run_async_migrations())


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

from app.core.config import get_settings가 되는 이유는 alembic.ini의 prepend_sys_path = . 덕분입니다. 프로젝트 루트를 import 경로에 넣어 주기 때문에, alembic 명령은 항상 프로젝트 루트(pyproject.toml이 있는 곳)에서 실행해야 합니다. 다른 폴더에서 실행하면 app을 못 찾습니다.
