① tests/conftest.py — 쿼리를 세는 fixture

3단계 파일 끝에 QueryCounter와 query_counter를 붙입니다. SQLAlchemy는 SQL을 DB로 보내기 직전에 before_cursor_execute라는 이벤트를 울립니다. 거기에 함수를 걸어 문장을 리스트에 모으면, 테스트가 "지금부터" reset()하고 요청을 보낸 뒤 selects를 세면 됩니다. async 엔진도 속은 동기 엔진(engine.sync_engine)이라 이벤트는 거기에 겁니다. 테스트가 끝나면 event.remove로 떼어 다른 테스트에 영향을 주지 않게 합니다.

async def _ensure_database_exists(database_url: str) -> None:
    """auction_test DB 가 없으면 만든다. CREATE DATABASE 는 트랜잭션 밖에서만 되므로 AUTOCOMMIT."""
    url = make_url(database_url)
    admin = create_async_engine(url.set(database="postgres"), isolation_level="AUTOCOMMIT")
    async with admin.connect() as conn:
        exists = await conn.scalar(
            text("SELECT 1 FROM pg_database WHERE datname = :name"), {"name": url.database}
        )
        if not exists:
            await conn.execute(text(f'CREATE DATABASE "{url.database}"'))
    await admin.dispose()

class QueryCounter:
    """실행된 SQL 문장을 모아 두는 작은 상자. N+1 을 '숫자'로 증명할 때 쓴다."""

    def __init__(self) -> None:
        self.statements: list[str] = []

    def record(self, statement: str) -> None:
        self.statements.append(statement)

    def reset(self) -> None:
        self.statements.clear()

    @property
    def selects(self) -> int:
        return sum(1 for s in self.statements if s.lstrip().upper().startswith("SELECT"))


@pytest.fixture
def query_counter() -> Iterator[QueryCounter]:
    """엔진이 커서에 SQL 을 보내기 직전(before_cursor_execute)마다 문장을 기록한다.
    async 엔진도 속은 동기 엔진(engine.sync_engine)이라 거기에 이벤트를 건다."""
    counter = QueryCounter()

    def _before_cursor_execute(conn, cursor, statement, parameters, context, executemany):  # noqa: ANN001
        counter.record(statement)

    event.listen(engine.sync_engine, "before_cursor_execute", _before_cursor_execute)
    yield counter
    event.remove(engine.sync_engine, "before_cursor_execute", _before_cursor_execute)

② tests/test_auction_list.py — 먼저 실패하는 테스트 14개

시드 25건의 규칙이 파일 맨 위 docstring에 있습니다. 마감 시각을 5건씩 같게 만든 것이 포인트입니다 — 동점이 있어야 "id로 가르는가"를 검사할 수 있습니다. _walk 헬퍼는 next_cursor가 null이 될 때까지 페이지를 넘기며 모읍니다. 테스트는 네 묶음입니다: 페이지 넘기기 4개(10/10/5, 정렬·동점, 기본 limit 20, 딱 맞아떨어질 때 커서 없음), 필터·정렬 4개, 잘못된 입력 4개(가격 역전, limit 범위, 엉터리 커서, 다른 정렬의 커서), N+1 증명 2개(정확히 2번 / 한 건씩 하면 21번).

pythontests/test_auction_list.py
"""GET /auctions — 커서 페이지네이션 · 필터 · 정렬 · N+1.

시드 데이터(25건)의 규칙
    - ends_at  : 5건씩 같은 시각(1시간 뒤, 2시간 뒤, …) → 동점을 id 로 가르는지 본다
    - created_at: i분 전 → 00번이 가장 최신
    - category : 짝수 hobby / 홀수 books
    - 가격     : 10,000 × (i+1) 원
    - status   : 7의 배수(0·7·14·21)는 ended, 나머지는 active
"""

import uuid
from datetime import UTC, datetime, timedelta

from httpx2 import AsyncClient
from sqlalchemy.ext.asyncio import AsyncSession

from app.models import Auction, AuctionCategory, AuctionStatus, User
from app.repositories import auction as auction_repo
from tests.conftest import QueryCounter
from tests.factories import create_auction, create_user

N = 25


async def _seed(db: AsyncSession, seller: User, n: int = N) -> list[Auction]:
    now = datetime.now(UTC).replace(microsecond=0)
    auctions = []
    for i in range(n):
        auctions.append(
            await create_auction(
                db,
                seller,
                title=f"경매 {i:02d}",
                ends_at=now + timedelta(hours=i // 5 + 1),
                created_at=now - timedelta(minutes=i),
                category=AuctionCategory.BOOKS if i % 2 else AuctionCategory.HOBBY,
                starting_price=10_000 * (i + 1),
                current_price=10_000 * (i + 1),
                status=AuctionStatus.ENDED if i % 7 == 0 else AuctionStatus.ACTIVE,
            )
        )
    return auctions


async def _walk(client: AsyncClient, **params: object) -> list[list[dict]]:
    """next_cursor 가 None 이 될 때까지 페이지를 넘기며 items 목록을 모은다."""
    pages: list[list[dict]] = []
    cursor: str | None = None
    while True:
        query = {k: v for k, v in params.items() if v is not None}
        if cursor:
            query["cursor"] = cursor
        res = await client.get("/auctions", params=query)
        assert res.status_code == 200, res.text
        body = res.json()
        pages.append(body["items"])
        cursor = body["next_cursor"]
        if cursor is None:
            return pages


# ── 페이지 넘기기 ─────────────────────────────────────────────────────


async def test_page_walk_covers_every_item_once(client: AsyncClient, db_session: AsyncSession):
    seeded = await _seed(db_session, await create_user(db_session))

    pages = await _walk(client, limit=10)

    assert [len(p) for p in pages] == [10, 10, 5]
    ids = [item["id"] for page in pages for item in page]
    assert len(ids) == len(set(ids)) == N  # 중복도 누락도 없다
    assert set(ids) == {str(a.id) for a in seeded}


async def test_ending_soon_is_sorted_by_ends_at_then_id(
    client: AsyncClient, db_session: AsyncSession
):

async def test_list_page_uses_exactly_two_queries(
    client: AsyncClient, db_session: AsyncSession, query_counter: QueryCounter
):
    """20건 + 판매자 20명 → 경매 SELECT 1번 + 판매자 IN (...) SELECT 1번 = 2번."""
    for i in range(20):
        await create_auction(db_session, await create_user(db_session, nickname=f"판매자{i}"))
    query_counter.reset()

    res = await client.get("/auctions", params={"limit": 20})

    assert res.status_code == 200 and len(res.json()["items"]) == 20
    assert query_counter.selects == 2, query_counter.statements


async def test_loading_seller_row_by_row_is_n_plus_one(
    db_session: AsyncSession, query_counter: QueryCounter
):
    """같은 20건을 selectinload 없이 가져와 한 건씩 판매자를 꺼내면 1 + 20 = 21번."""
    for i in range(20):
        await create_auction(db_session, await create_user(db_session, nickname=f"판매자{i}"))
    db_session.expunge_all()  # 세션이 기억하는 객체를 비워 '처음 보는 데이터' 상황을 만든다
    query_counter.reset()

    rows = await auction_repo.list_page(db_session, limit=20, load_seller=False)
    assert query_counter.selects == 1
    for auction in rows[:20]:
        await db_session.run_sync(lambda _s, a=auction: a.seller.nickname)  # 동기 ORM 의 a.seller

    assert query_counter.selects == 1 + 20
마지막 테스트의 expunge_all(). 시드를 만든 세션은 판매자 20명을 이미 기억(identity map)하고 있어서, 그냥 a.seller를 부르면 DB에 가지 않고 메모리에서 꺼냅니다. "처음 보는 데이터"라는 실제 요청 상황을 흉내 내려고 세션의 기억을 비운 뒤 셉니다. run_sync(lambda: a.seller)는 동기 ORM에서 a.seller라고 쓰는 것과 같습니다 — async 세션에서는 이렇게 감싸지 않으면 MissingGreenlet 에러가 납니다(함정 절 참고).

④ app/core/exceptions.py — 예외 하나 추가

커서가 깨졌을 때 던질 InvalidCursorError입니다. 부모가 UnprocessableError라 422와 껍데기가 자동입니다. 4단계에서 "새 규칙은 부모만 고르면 된다"고 한 것이 그대로 적용되는 첫 사례입니다.

python(수정) app/core/exceptions.py
"""도메인 예외와 표준 에러 응답.

서비스 계층은 HTTP 를 모른다 — HTTPException 대신 여기 있는 AppError 계열만 던진다.
라우터 바깥의 핸들러가 AppError 를 {"code", "message"} 형태의 JSON 으로 번역한다.

계층
    AppError (400)
    ├── UnauthorizedError (401)
    ├── ForbiddenError (403)      ── NotAuctionSellerError
    ├── NotFoundError (404)       ── AuctionNotFoundError
    ├── ConflictError (409)       ── EmailAlreadyExistsError, AuctionHasBidsError
    │   └── InvalidStateError     ── AuctionClosedError
    └── UnprocessableError (422)  ── InvalidPeriodError, InvalidCursorError
FastAPI 가 내는 422(RequestValidationError) 와 404/405(StarletteHTTPException) 도

async def app_error_handler(request: Request, exc: AppError) -> JSONResponse:
    headers = {"WWW-Authenticate": "Bearer"} if exc.status_code == 401 else None
    return JSONResponse(
        status_code=exc.status_code,
        content={"code": exc.code, "message": exc.message},
        headers=headers,
    )


async def validation_error_handler(request: Request, exc: RequestValidationError) -> JSONResponse:
    """Pydantic 검증 실패(422). 기본 응답은 {"detail": [...]} 인데, 우리 껍데기로 감싸고
    필드별 상세는 "errors" 에 넣는다 — 프론트가 입력칸 옆에 메시지를 보여줄 때 쓴다.
    (loc / msg / type 만 남긴다. 기본 목록에 딸려 오는 input·ctx 는 사용자 입력을 되돌려 주거나
    파이썬 객체라 JSON 으로 못 만드는 경우가 있어 뺀다.)"""
    errors = [{"loc": e["loc"], "msg": e["msg"], "type": e["type"]} for e in exc.errors()]
    return JSONResponse(
        status_code=status.HTTP_422_UNPROCESSABLE_CONTENT,
        content={
            "code": "VALIDATION_ERROR",
            "message": "요청 값이 올바르지 않습니다",
            "errors": errors,
        },
    )


async def http_exception_handler(request: Request, exc: StarletteHTTPException) -> JSONResponse:
    """라우트 없음(404)·허용되지 않은 메서드(405) 같은 프레임워크 기본 에러도 같은 껍데기로."""
    codes = {404: "NOT_FOUND", 405: "METHOD_NOT_ALLOWED"}
    code = codes.get(exc.status_code, f"HTTP_{exc.status_code}")
    return JSONResponse(
        status_code=exc.status_code,
        content={"code": code, "message": exc.detail},
        headers=exc.headers,
    )


def register_exception_handlers(app: FastAPI) -> None:
    app.add_exception_handler(AppError, app_error_handler)  # type: ignore[arg-type]
    app.add_exception_handler(RequestValidationError, validation_error_handler)  # type: ignore[arg-type]
    app.add_exception_handler(StarletteHTTPException, http_exception_handler)  # type: ignore[arg-type]

⑤ app/core/pagination.py — 커서 만들기·풀기

순수 함수 둘뿐입니다. encode_cursor는 [정렬 기준, 시각의 ISO 문자열, id]를 JSON으로 만들고 base64url로 감싼 뒤 끝의 = 패딩을 뗍니다(URL에 그대로 넣기 좋게). decode_cursor는 그 반대인데, 어디서 실패하든 — base64가 아니든, JSON이 아니든, 항목이 셋이 아니든, 시각이나 UUID 모양이 아니든 — 전부 InvalidCursorError 하나로 모읍니다. 클라이언트가 커서를 만들 일은 없으니 "뭐가 틀렸는지" 친절할 필요가 없고, "처음부터 다시 조회하세요"면 충분합니다.

정렬 기준을 커서 안에 넣는 이유는 마지막 줄입니다. sort=newest로 받은 커서를 sort=ending_soon에 쓰면 에러 없이 엉뚱한 곳부터 읽힙니다. 조용한 오동작보다 422가 낫습니다.

pythonapp/core/pagination.py
"""커서(keyset) 페이지네이션 도구.

커서 = "마지막으로 본 항목의 정렬 키" 를 base64 로 싼 문자열.
    [정렬 기준, 정렬 키(ISO 시각), id]  →  json  →  base64url (패딩 제거)
클라이언트는 내용을 해석하지 않고 그대로 되돌려 준다(불투명 토큰).
정렬 기준을 같이 넣는 이유: newest 로 받은 커서를 ending_soon 에 쓰면 엉뚱한 곳부터 읽힌다.
"""

import base64
import json
import uuid
from datetime import datetime

from app.core.exceptions import InvalidCursorError


def encode_cursor(sort: str, key: datetime, id_: uuid.UUID) -> str:
    raw = json.dumps([sort, key.isoformat(), str(id_)]).encode()
    return base64.urlsafe_b64encode(raw).decode().rstrip("=")


def decode_cursor(cursor: str, sort: str) -> tuple[datetime, uuid.UUID]:
    """되돌려 받은 커서를 (정렬 키, id) 로. 모양이 조금이라도 다르면 InvalidCursorError."""
    try:
        padded = cursor + "=" * (-len(cursor) % 4)  # 떼어 낸 '=' 패딩을 되살린다
        sort_in, key, id_ = json.loads(base64.urlsafe_b64decode(padded))
        parsed = (datetime.fromisoformat(key), uuid.UUID(id_))
    except (ValueError, TypeError) as exc:  # base64·JSON·개수·시각·UUID 오류 전부 여기로
        raise InvalidCursorError() from exc
    if sort_in != sort:
        raise InvalidCursorError("정렬 기준이 다른 커서입니다")
    return parsed

⑦ app/repositories/auction.py — list_page

SQL이 사는 곳입니다. 필터는 있는 것만 where로 쌓고, 커서(after)가 있으면 tuple_(key, Auction.id)로 행 비교를 합니다 — 렌더링되면 (ends_at, id) > ($1, $2)입니다. 오름차순(ending_soon)이면 >, 내림차순(newest)이면 <. ORDER BY는 반드시 커서 키와 같은 두 컬럼, 같은 방향이어야 합니다(다르면 페이지가 꼬입니다). selectinload(Auction.seller) 한 줄이 두 번째 그림의 오른쪽을 만들고, limit + 1이 "다음 있음"의 재료입니다.

"""auctions 테이블 접근. 쿼리만 있고 규칙은 없다."""

import uuid
from datetime import datetime

from sqlalchemy import select, tuple_
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.orm import selectinload

from app.models import Auction, AuctionCategory, AuctionStatus


async def get(db: AsyncSession, auction_id: uuid.UUID) -> Auction | None:
    return await db.get(Auction, auction_id)


async def add(db: AsyncSession, auction: Auction) -> Auction:
    db.add(auction)
    await db.flush()  # INSERT → id / created_at 확보. 커밋은 서비스가 결정한다
    return auction


async def list_page(
    db: AsyncSession,
    *,
    status: AuctionStatus | None = None,
    category: AuctionCategory | None = None,
    min_price: int | None = None,
    max_price: int | None = None,
    sort: str = "ending_soon",
    limit: int = 20,
    after: tuple[datetime, uuid.UUID] | None = None,
    load_seller: bool = True,
) -> list[Auction]:
    """keyset 페이지 한 장. limit+1 건을 가져와 '다음 페이지가 있나' 를 알려 준다.
    (limit 개로 잘라내는 일은 서비스가 한다.)

    정렬 키는 (시각, id) 쌍이다. 시각이 같은 행이 여러 개여도 id 가 순서를 확정하므로
    페이지 경계에서 같은 행이 두 번 나오거나 빠지는 일이 없다.
    """
    key = Auction.ends_at if sort == "ending_soon" else Auction.created_at
    ascending = sort == "ending_soon"

    stmt = select(Auction)
    if status is not None:
        stmt = stmt.where(Auction.status == status)
    if category is not None:
        stmt = stmt.where(Auction.category == category)
    if min_price is not None:
        stmt = stmt.where(Auction.current_price >= min_price)
    if max_price is not None:
        stmt = stmt.where(Auction.current_price <= max_price)

    if after is not None:
        pair = tuple_(key, Auction.id)  # SQL: (ends_at, id) > ($1, $2)  ← 행 비교
        stmt = stmt.where(pair > after if ascending else pair < after)

    if ascending:
        stmt = stmt.order_by(key.asc(), Auction.id.asc())
    else:
        stmt = stmt.order_by(key.desc(), Auction.id.desc())

    if load_seller:
        stmt = stmt.options(selectinload(Auction.seller))  # 판매자는 IN (...) 한 방으로

    result = await db.scalars(stmt.limit(limit + 1))
    return list(result.all())

import uuid
from datetime import UTC, datetime

from sqlalchemy.ext.asyncio import AsyncSession

from app.core.exceptions import (
    AuctionClosedError,
    AuctionHasBidsError,
    AuctionNotFoundError,
    InvalidPeriodError,
    NotAuctionSellerError,
)
from app.core.pagination import decode_cursor, encode_cursor
from app.models import Auction, AuctionStatus, User
from app.repositories import auction as auction_repo
from app.schemas.auction import AuctionCreate, AuctionListParams, AuctionUpdate

OPEN_STATUSES = {AuctionStatus.SCHEDULED, AuctionStatus.ACTIVE}


def _now() -> datetime:
    return datetime.now(UTC)


async def create(db: AsyncSession, seller: User, data: AuctionCreate) -> Auction:
    now = _now()
    starts_at = data.starts_at or now
    auction = Auction(
        seller_id=seller.id,
        title=data.title,
        description=data.description,
        category=data.category,
        starting_price=data.starting_price,
        current_price=data.starting_price,  # 입찰 전 현재가 = 시작가
        min_increment=data.min_increment,
        starts_at=starts_at,
        ends_at=data.ends_at,
        status=AuctionStatus.ACTIVE if starts_at <= now else AuctionStatus.SCHEDULED,
    )
    await auction_repo.add(db, auction)
    await db.commit()
    await db.refresh(auction)  # DB 가 채운 created_at / updated_at 을 읽어 온다
    return auction


async def list_auctions(
    db: AsyncSession, params: AuctionListParams
) -> tuple[list[Auction], str | None]:
    """한 페이지 + 다음 커서. 커서 해석/생성은 여기서, 쿼리는 리포지토리에서."""
    after = decode_cursor(params.cursor, params.sort) if params.cursor else None
    rows = await auction_repo.list_page(
        db,
        status=params.status,
        category=params.category,
        min_price=params.min_price,
        max_price=params.max_price,
        sort=params.sort,
        limit=params.limit,
        after=after,
    )
    items = rows[: params.limit]
    if len(rows) <= params.limit:  # limit+1 번째가 없다 = 마지막 페이지
        return items, None
    last = items[-1]
    key = last.ends_at if params.sort == "ending_soon" else last.created_at
    return items, encode_cursor(params.sort, key, last.id)


async def get(db: AsyncSession, auction_id: uuid.UUID) -> Auction:
    auction = await auction_repo.get(db, auction_id)
    if auction is None:
        raise AuctionNotFoundError()
    return auction


def _ensure_seller(auction: Auction, user: User) -> None:
    if auction.seller_id != user.id:
        raise NotAuctionSellerError()


def _ensure_open_and_unbid(auction: Auction) -> None:
    """수정·취소 공통 규칙: 아직 열려 있고(scheduled/active), 입찰이 하나도 없어야 한다."""
    if auction.status not in OPEN_STATUSES:
        raise AuctionClosedError()
    if auction.bid_count > 0:
        raise AuctionHasBidsError()


async def update(
    db: AsyncSession, user: User, auction_id: uuid.UUID, data: AuctionUpdate
) -> Auction:
    auction = await get(db, auction_id)
    _ensure_seller(auction, user)
    _ensure_open_and_unbid(auction)

    changes = data.model_dump(exclude_unset=True)  # 요청에 실제로 담긴 필드만
    if "ends_at" in changes and changes["ends_at"] <= auction.starts_at:
        raise InvalidPeriodError()
    for field, value in changes.items():
        setattr(auction, field, value)
    auction.version += 1  # 6단계 낙관적 락의 재료 — 바뀔 때마다 1 증가

    await db.commit()
    await db.refresh(auction)
    return auction


async def cancel(db: AsyncSession, user: User, auction_id: uuid.UUID) -> Auction:
    auction = await get(db, auction_id)
    _ensure_seller(auction, user)
    _ensure_open_and_unbid(auction)

    auction.status = AuctionStatus.CANCELLED
    auction.version += 1
    await db.commit()
    await db.refresh(auction)
    return auction
