1. tests/test_closer.py — 시계를 쥔 테스트

일곱 개입니다. 앞의 다섯은 롤백 세션(db_session)으로 충분합니다 — 서비스가 commit() 해도 3단계 fixture 가 SAVEPOINT 로 받아 주고 끝나면 전부 롤백됩니다. 낙찰(winner·Payment·이벤트 세 가지를 한 번에), 유찰(winner 없음·Payment 없음·이벤트는 감), 마감 전·취소된 경매는 건드리지 않음, 활성화(두 번 돌려도 0), 결제 만료(기한 지난 pending 만). 이벤트는 events.publish 를 monkeypatch 로 기록기로 바꿔 어떤 payload 가 나갔는지 결정적으로 봅니다.

뒤의 둘은 @pytest.mark.commits — 진짜 커밋합니다. 잠금은 서로 다른 커넥션 사이에서만 의미가 있어서, 워커 두 개를 흉내 내려면 세션 두 개로 정말 커밋해야 합니다(6단계와 같은 이유). 마감된 경매 5건을 두 코루틴이 동시에 close_due 하면 합쳐서 정확히 5건, 중복 0, payments 5행. 마지막은 루프 자체: run() 이 interval=0.05 로 돌다가 stop 이 set 되면 돌아오는지, 첫 틱이 5건을 닫고 그다음 틱들은 빈손인지.

async def _bid(db: AsyncSession, auction: Auction, bidder: User, amount: int, at: datetime) -> Bid:
    """입찰 행을 직접 만든다 (place_bid 의 락·rate limit 은 6단계가 검증했다)."""
    bid = Bid(auction_id=auction.id, bidder_id=bidder.id, amount=amount, created_at=at)
    db.add(bid)
    auction.current_price = amount
    auction.bid_count += 1
    await db.flush()
    return bid

    assert closed[0].payment_id == payment.id

    assert [e.type for e in published] == ["auction_closed"]
    assert published[0].auction_id == auction.id
    assert published[0].payload["winner_id"] == carol.id
    assert published[0].payload["final_price"] == 12_000

@pytest.mark.commits
async def test_two_workers_close_each_auction_exactly_once(
    five_due_auctions: list[uuid.UUID], redis: Redis
):
    async def worker() -> list[closer_service.ClosedResult]:

        )
        payments = await db.scalar(select(func.count()).select_from(Payment))
    assert ended == 5 and payments == 5


@pytest.mark.commits
async def test_worker_loop_ticks_and_stops_cleanly(
    five_due_auctions: list[uuid.UUID], redis: Redis, monkeypatch: pytest.MonkeyPatch
):
    """run() 은 stop 이 set 될 때까지 interval 마다 tick 하고, set 되면 조용히 돌아온다."""
    ticks: list[closer_worker.TickResult] = []
    original_tick = closer_worker.tick

    async def recording_tick(redis: Redis, now: datetime | None = None) -> closer_worker.TickResult:
        result = await original_tick(redis, now=NOW)  # 시계를 고정
        ticks.append(result)
        return result

    monkeypatch.setattr(closer_worker, "tick", recording_tick)
    stop = asyncio.Event()

    async def stop_soon() -> None:
        await asyncio.sleep(0.3)
        stop.set()

    await asyncio.gather(closer_worker.run(stop, redis, interval=0.05), stop_soon())

    assert len(ticks) >= 2
    assert len(ticks[0].closed) == 5 and ticks[0].activated == 0 and ticks[0].expired == 0
    assert all(t.empty for t in ticks[1:])  # 두 번째 틱부터는 할 일이 없다

4. app/repositories/auction.py — skip_locked 와 후보 조회 (수정)

6단계의 get_for_update 에 skip_locked 스위치를 답니다. 기본값 False 라 입찰 쪽은 한 글자도 안 바뀝니다. 그리고 "시작 시각 지난 scheduled"·"마감 시각 지난 active" 의 id 만 뽑는 조회 두 개 — 락 없이, limit 만큼. 마감 후보 조회는 2단계에서 만든 ix_auctions_status_ends_at 인덱스를 정확히 탑니다(WHERE status = … AND ends_at <= …).

왜 id 만 뽑고 다시 잠그나. SELECT * … FOR UPDATE SKIP LOCKED LIMIT 100 한 방으로 100건을 잠그고 처리할 수도 있습니다. 하지만 그러면 100건이 한 트랜잭션이 되어 (1) 99번째에서 실패하면 전부 롤백, (2) 100개 행이 처리가 끝날 때까지 잠겨 그 사이 입찰 요청이 줄을 섭니다(마감 직후 입찰은 어차피 거절이지만, 락 대기 자체가 응답을 늦춥니다). 후보는 값싸게 보고, 잠금은 한 건씩 짧게 — 큐 워커의 정석입니다.

"""auctions 테이블 접근. 쿼리만 있고 규칙은 없다."""

import uuid
from datetime import datetime

from sqlalchemy import select, tuple_
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.orm import selectinload

from app.models import Auction, AuctionCategory, AuctionStatus


async def get(db: AsyncSession, auction_id: uuid.UUID) -> Auction | None:
    return await db.get(Auction, auction_id)


async def get_for_update(
    db: AsyncSession, auction_id: uuid.UUID, *, skip_locked: bool = False
) -> Auction | None:
    """행 잠금(SELECT … FOR UPDATE). 이 트랜잭션이 커밋/롤백될 때까지 같은 행을 잠그려는
    다른 트랜잭션은 줄을 선다 → '읽고 → 판단하고 → 쓰는' 사이에 아무도 끼어들지 못한다.

    populate_existing: 이 세션이 같은 행을 이미 읽어 둔 적이 있어도 DB 의 최신 값으로 덮어쓴다.
    (락을 기다리는 동안 누가 값을 바꿨을 수 있다 — 옛 값으로 판단하면 락을 건 의미가 없다.)

    skip_locked=True (8단계): 줄을 서지 않고, 이미 남이 잠근 행이면 None 을 돌려준다.
    입찰(6단계)은 "기다렸다가 내 차례에 처리" 가 맞지만,
    마감 워커는 "남이 하고 있으면 나는 다음 것" 이 맞다.
    """
    stmt = (
        select(Auction)
        .where(Auction.id == auction_id)
        .with_for_update(skip_locked=skip_locked)
        .execution_options(populate_existing=True)
    )
    return await db.scalar(stmt)


async def list_due_scheduled_ids(db: AsyncSession, now: datetime, *, limit: int) -> list[uuid.UUID]:
    """시작 시각이 지난 scheduled 경매 id — 활성화 후보. 락 없음(값싼 조회)."""
    stmt = (
        select(Auction.id)
        .where(Auction.status == AuctionStatus.SCHEDULED, Auction.starts_at <= now)
        .order_by(Auction.starts_at)
        .limit(limit)
    )
    return list((await db.scalars(stmt)).all())


async def list_due_active_ids(db: AsyncSession, now: datetime, *, limit: int) -> list[uuid.UUID]:
    """마감 시각이 지난 active 경매 id — 마감 후보. ix_auctions_status_ends_at 인덱스를 탄다."""
    stmt = (
        select(Auction.id)
        .where(Auction.status == AuctionStatus.ACTIVE, Auction.ends_at <= now)
        .order_by(Auction.ends_at)
        .limit(limit)
    )
    return list((await db.scalars(stmt)).all())


async def add(db: AsyncSession, auction: Auction) -> Auction:
    db.add(auction)
    await db.flush()  # INSERT → id / created_at 확보. 커밋은 서비스가 결정한다
    return auction


async def list_page(
    db: AsyncSession,
    *,
    status: AuctionStatus | None = None,
    category: AuctionCategory | None = None,
    min_price: int | None = None,
    max_price: int | None = None,
    sort: str = "ending_soon",
    limit: int = 20,
    after: tuple[datetime, uuid.UUID] | None = None,
    load_seller: bool = True,
) -> list[Auction]:
    """keyset 페이지 한 장. limit+1 건을 가져와 '다음 페이지가 있나' 를 알려 준다.
    (limit 개로 잘라내는 일은 서비스가 한다.)

    정렬 키는 (시각, id) 쌍이다. 시각이 같은 행이 여러 개여도 id 가 순서를 확정하므로
    페이지 경계에서 같은 행이 두 번 나오거나 빠지는 일이 없다.
    """
    key = Auction.ends_at if sort == "ending_soon" else Auction.created_at
    ascending = sort == "ending_soon"

    stmt = select(Auction)
    if status is not None:
        stmt = stmt.where(Auction.status == status)
    if category is not None:
        stmt = stmt.where(Auction.category == category)
    if min_price is not None:
        stmt = stmt.where(Auction.current_price >= min_price)
    if max_price is not None:
        stmt = stmt.where(Auction.current_price <= max_price)

    if after is not None:
        pair = tuple_(key, Auction.id)  # SQL: (ends_at, id) > ($1, $2)  ← 행 비교
        stmt = stmt.where(pair > after if ascending else pair < after)

    if ascending:
        stmt = stmt.order_by(key.asc(), Auction.id.asc())
    else:
        stmt = stmt.order_by(key.desc(), Auction.id.desc())

    if load_seller:
        stmt = stmt.options(selectinload(Auction.seller))  # 판매자는 IN (...) 한 방으로

    result = await db.scalars(stmt.limit(limit + 1))
    return list(result.all())

7. app/services/closer.py — 규칙 세 가지, now 는 인자

핵심 파일입니다. 세 함수 모두 HTTP·루프·시그널을 모르고, now 를 받습니다. activate_due 와 expire_payments 는 update(...).where(...).values(...).returning(id) 한 문장 + 커밋 — UPDATE 가 잠근 행은 커밋까지 다른 UPDATE 를 기다리게 하고, 기다린 쪽은 커밋된 최신 값으로 WHERE 를 다시 평가하므로(READ COMMITTED) 이미 active 인 행은 0 건으로 빠집니다. 워커가 둘이어도 안전한 이유입니다.

close_due 는 설계 그림 그대로입니다. 후보 id 목록 → 한 건씩 get_for_update(skip_locked=True) → None 이면 남의 몫 → 잠근 뒤 status·ends_at 재검사(아니면 rollback() 으로 락만 풀고 다음) → top_bid 로 낙찰자 → ended·winner_id·version+1 → 입찰이 있었으면 Payment(pending, expires_at = now + 24h) → commit() → 그리고 커밋 뒤에 auction_closed 발행과 [notify] 로그. 이메일 발송은 이 프로젝트의 비목표라 로그 한 줄이 알림을 대신합니다. 진짜라면 그 자리에 "알림 큐에 넣기"가 옵니다.

pythonapp/services/closer.py
"""시간이 흐르면 저절로 일어나야 하는 일 세 가지. HTTP 도, 루프도, 시그널도 모른다.

  activate_due   : 시작 시각이 지난 scheduled 경매 → active
  close_due      : 마감 시각이 지난 active 경매 → ended (+ 낙찰자, Payment, auction_closed 이벤트)
  expire_payments: 결제 기한이 지난 pending 결제 → expired

세 함수 모두 `now` 를 인자로 받는다. 워커는 진짜 시계를 넣고, 테스트는 원하는 시각을 넣는다.
누가 부르든(워커 1개든 2개든) 결과가 같아야 한다 — 그게 이 파일의 유일한 어려움이다.
"""

import logging
import uuid
from dataclasses import dataclass
from datetime import datetime, timedelta

from redis.asyncio import Redis
from sqlalchemy import update
from sqlalchemy.ext.asyncio import AsyncSession

from app.core import events
from app.core.config import get_settings
from app.models import Auction, AuctionStatus, Payment, PaymentStatus
from app.repositories import auction as auction_repo
from app.repositories import bid as bid_repo
from app.repositories import payment as payment_repo

logger = logging.getLogger(__name__)


@dataclass(frozen=True)
class ClosedResult:
    auction_id: uuid.UUID
    winner_id: uuid.UUID | None  # None = 유찰 (입찰 0건)
    final_price: int | None
    bid_count: int
    payment_id: uuid.UUID | None


async def activate_due(db: AsyncSession, now: datetime, *, limit: int | None = None) -> int:
    """UPDATE 한 문장. UPDATE 는 스스로 행을 잠그므로 워커가 둘이어도 안전하다 —
    둘째 워커는 첫째의 커밋을 기다렸다가 다시 조건을 보고(이미 active) 0 행을 바꾼다."""
    limit = limit or get_settings().closer_batch_size
    ids = await auction_repo.list_due_scheduled_ids(db, now, limit=limit)
    if not ids:
        return 0
    stmt = (
        update(Auction)
        .where(Auction.id.in_(ids), Auction.status == AuctionStatus.SCHEDULED)  # 조건을 한 번 더
        .values(status=AuctionStatus.ACTIVE, version=Auction.version + 1)
        .returning(Auction.id)
    )
    activated = list((await db.scalars(stmt)).all())
    await db.commit()
    if activated:
        logger.info("activated %d auction(s): %s", len(activated), [str(i)[:8] for i in activated])
    return len(activated)


async def close_due(
    db: AsyncSession, redis: Redis, now: datetime, *, limit: int | None = None
) -> list[ClosedResult]:
    """후보 목록은 락 없이 값싸게 뽑고, 실제 처리는 **한 건씩** 잠그고 커밋한다.

    왜 한 건씩인가: 100건을 한 트랜잭션에 묶으면 (1) 100번째에서 실패하면 99건이 같이 롤백되고,
    (2) 그동안 100개 행이 잠겨 입찰 요청들이 줄을 선다. 한 건씩이면 실패도 잠금도 한 건뿐이다.
    왜 skip_locked 인가: 다른 워커가 이미 잡은 행이면 기다릴 이유가 없다 — 그쪽이 끝낼 것이다.
    왜 재검사인가: 후보를 뽑은 뒤 잠그기까지 사이에 누가 이미 닫았을 수 있다(그 워커가 커밋해서
    락을 풀면 우리는 status=ended 인 행을 잠그게 된다). 잠근 *뒤에* 본 값만 믿는다.
    """
    settings = get_settings()
    limit = limit or settings.closer_batch_size
    results: list[ClosedResult] = []

    for auction_id in await auction_repo.list_due_active_ids(db, now, limit=limit):
        auction = await auction_repo.get_for_update(db, auction_id, skip_locked=True)
        if auction is None:  # 다른 워커가 잠갔거나(skip) 사라졌다 → 내 몫이 아니다
            continue
        if auction.status != AuctionStatus.ACTIVE or auction.ends_at > now:  # 잠근 뒤 재검사
            await db.rollback()  # 잠금만 풀고 다음
            continue

        top = await bid_repo.top_bid(db, auction.id)
        auction.status = AuctionStatus.ENDED
        auction.winner_id = top.bidder_id if top else None
        auction.version += 1

        payment: Payment | None = None
        if top is not None:
            payment = await payment_repo.add(
                db,
                Payment(
                    auction_id=auction.id,
                    payer_id=top.bidder_id,
                    amount=auction.current_price,
                    status=PaymentStatus.PENDING,
                    expires_at=now + timedelta(hours=settings.payment_deadline_hours),
                ),
            )
        await db.commit()  # ← 여기서 이 경매의 락이 풀린다. 아래는 커밋된 사실만 방송한다

        result = ClosedResult(
            auction_id=auction.id,
            winner_id=auction.winner_id,
            final_price=auction.current_price if top else None,
            bid_count=auction.bid_count,
            payment_id=payment.id if payment else None,
        )
        await events.publish(
            redis,
            events.AuctionEvent(
                type="auction_closed",
                auction_id=auction.id,
                payload={
                    "winner_id": result.winner_id,
                    "final_price": result.final_price,
                    "bid_count": result.bid_count,
                    "payment_id": result.payment_id,
                    "payment_expires_at": payment.expires_at if payment else None,
                    "version": auction.version,
                },
            ),
        )
        if top is not None:
            # 진짜라면 여기서 메일/푸시 큐에 넣는다. 이 프로젝트에서는 로그 한 줄이 '알림' 이다.
            logger.info(
                "[notify] user=%s 낙찰! auction=%s %s원 — %s 까지 결제해 주세요",
                str(top.bidder_id)[:8],
                str(auction.id)[:8],
                f"{auction.current_price:,}",
                payment.expires_at.isoformat(timespec="minutes") if payment else "",
            )
        else:
            logger.info("[notify] auction=%s 유찰 (입찰 0건)", str(auction.id)[:8])
        results.append(result)
    return results


async def expire_payments(db: AsyncSession, now: datetime) -> int:
    """기한 지난 pending → expired. activate_due 와 같은 'UPDATE 한 문장' 패턴."""
    stmt = (
        update(Payment)
        .where(Payment.status == PaymentStatus.PENDING, Payment.expires_at <= now)
        .values(status=PaymentStatus.EXPIRED)
        .returning(Payment.id)
    )
    expired = list((await db.scalars(stmt)).all())
    await db.commit()
    if expired:
        logger.info("expired %d payment(s): %s", len(expired), [str(i)[:8] for i in expired])
    return len(expired)

8. app/workers/closer.py — 루프, 시그널, 로그

규칙은 위에 다 있으니 여기엔 "언제 부르나·어떻게 멈추나"만 있습니다. tick() 은 함수마다 SessionLocal() 을 새로 엽니다 — 하나가 실패해도 나머지 둘은 독립적으로 커밋되고, 틱 사이에 커넥션을 쥐고 있지 않습니다. run() 은 stop 이벤트가 set 될 때까지 돌고, 한 틱이 예외로 죽어도 logger.exception 만 남기고 계속 돕니다(워커는 한 번의 실패로 죽으면 안 됩니다). 기다릴 땐 asyncio.sleep(interval) 이 아니라 wait_for(stop.wait(), timeout=interval) — 종료 신호가 오면 interval 을 다 채우지 않고 즉시 깹니다. main() 은 SIGINT/SIGTERM 을 stop.set 에 연결하고, 끝나면 Redis·엔진을 닫습니다. 워커는 자기 Redis 클라이언트를 만듭니다 — API 프로세스의 redis_client 는 다른 프로세스의 것이니까요.

pythonapp/workers/closer.py
"""마감 워커 — API 서버와 **별개의 프로세스**.  실행:  uv run python -m app.workers.closer

하는 일은 단순하다: interval 초마다 시계를 보고 services/closer 의 세 함수를 차례로 부른다.
규칙은 전부 서비스에 있고, 이 파일은 "언제 부르나 · 어떻게 멈추나 · 무엇을 남기나" 만 안다.

왜 API 안의 BackgroundTasks 나 asyncio.create_task 가 아닌가:
  - 요청이 없어도 돌아야 한다 (새벽 3시에 마감되는 경매도 제때 닫혀야 한다)
  - uvicorn --workers 4 면 태스크도 4개가 되고, 서버를 재시작하면 사라진다
  - 별도 프로세스면 API 와 독립적으로 죽이고 살리고 늘릴 수 있다 (10단계 compose 의 `worker` 서비스)
"""

import asyncio
import logging
import signal
from dataclasses import dataclass, field
from datetime import UTC, datetime

from redis.asyncio import Redis

from app.core.config import get_settings
from app.core.db import SessionLocal, engine
from app.services import closer
from app.services.closer import ClosedResult

logger = logging.getLogger("worker.closer")


@dataclass
class TickResult:
    activated: int = 0
    closed: list[ClosedResult] = field(default_factory=list)
    expired: int = 0

    @property
    def empty(self) -> bool:
        return self.activated == 0 and not self.closed and self.expired == 0


async def tick(redis: Redis, now: datetime | None = None) -> TickResult:
    """한 번의 '시계 보기'. 함수마다 세션을 새로 여는 이유: 하나가 실패해도 다른 둘은 독립적으로
    커밋되고, 세션(커넥션)을 틱 사이에 쥐고 있지 않기 위해서다."""
    now = now or datetime.now(UTC)
    result = TickResult()
    async with SessionLocal() as db:
        result.activated = await closer.activate_due(db, now)
    async with SessionLocal() as db:
        result.closed = await closer.close_due(db, redis, now)
    async with SessionLocal() as db:
        result.expired = await closer.expire_payments(db, now)
    return result


async def run(stop: asyncio.Event, redis: Redis, interval: float) -> None:
    """stop 이 set 될 때까지 interval 마다 tick. 한 틱이 실패해도 루프는 죽지 않는다(로그만)."""
    logger.info("closer started (interval=%ss)", interval)
    while not stop.is_set():
        try:
            result = await tick(redis)
            if not result.empty:
                logger.info(
                    "tick: activated=%d closed=%d expired=%d",
                    result.activated,
                    len(result.closed),
                    result.expired,
                )
        except Exception:  # noqa: BLE001 — 워커는 한 틱의 실패로 죽으면 안 된다
            logger.exception("tick failed")
        try:
            # sleep 하면 종료 신호를 interval 만큼 늦게 받는다. stop.wait() 를 기다리면 즉시 깬다.
            await asyncio.wait_for(stop.wait(), timeout=interval)
        except TimeoutError:
            pass
    logger.info("closer stopped")


async def main() -> None:
    settings = get_settings()
    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s [pid %(process)d] %(levelname)s %(name)s: %(message)s",
    )
    redis = Redis.from_url(settings.redis_url, decode_responses=True)  # 워커 전용 클라이언트
    stop = asyncio.Event()
    loop = asyncio.get_running_loop()
    for sig in (signal.SIGINT, signal.SIGTERM):  # Ctrl-C, docker stop → 조용히 마무리
        loop.add_signal_handler(sig, stop.set)
    try:
        await run(stop, redis, settings.closer_interval_seconds)
    finally:
        await redis.aclose()
        await engine.dispose()


if __name__ == "__main__":
    asyncio.run(main())
