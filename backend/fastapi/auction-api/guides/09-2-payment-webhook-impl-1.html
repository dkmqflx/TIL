1. tests/test_payments.py — 두 방향을 전부 연출한다

열여섯 개입니다. 전부 롤백 세션(db_session)으로 됩니다 — 이번엔 동시성 테스트가 없습니다(멱등성은 DB 의 PK 충돌이 보장하므로, "같은 event_id 두 번" 을 순서대로 보내는 것으로 충분합니다). pg_mock fixture 가 respx.mock(base_url=…) 를 켜면, 그 테스트 동안 http.post("http://localhost:8001/…") 는 네트워크로 나가지 않고 우리가 정한 응답을 받습니다. route.call_count 로 "몇 번 두드렸나" 를 셉니다 — 재시도 테스트의 핵심 숫자입니다. 웹훅 쪽은 signed() 헬퍼가 본문을 json.dumps 한 바이트에 서명해 헤더로 만들고, tamper= 로 "서명은 원본, 본문만 바꿔치기" 를 연출합니다.

checkout 7개: 성공(PG 에 보낸 본문·callback_url 까지 확인), 두 번 눌러도 PG 1회, 남이 누르면 403, 기한 지남 409 PAYMENT_EXPIRED, 이미 paid 409 PAYMENT_NOT_PENDING, 타임아웃 3회 → 502 + external_id 없음, 500 뒤 200 → 성공·2회. 웹훅 7개: 서명 없음 401, 본문 변조 401(상태 그대로), 정상 → paid·paid_at·이벤트 1건, 같은 event_id 두 번 → duplicate: true·paid_at 그대로·이벤트 1건, 모르는 external_id 404, 금액 불일치 422, failed → external_id 비움(재시도 가능). 조회·만료 2개: payer·판매자만 200 / 남 403 / 없는 id 404, 워커가 만료시킨 뒤 checkout → 409 PAYMENT_NOT_PENDING.

async def won_payment(
    db: AsyncSession,
    *,
    seller: User,
    payer: User,
    amount: int = 12_000,
    expires_in: timedelta = timedelta(hours=24),
    status: PaymentStatus = PaymentStatus.PENDING,
    external_id: str | None = None,
) -> tuple[Auction, Payment]:
    """낙찰이 끝난 경매 + 결제 대기표. 8단계 close_due 가 만드는 것과 같은 모양을 직접 넣는다."""
    now = datetime.now(UTC)
    auction = await create_auction(
        db,
        seller,
        status=AuctionStatus.ENDED,
        starts_at=now - timedelta(days=2),
        ends_at=now - timedelta(days=1),
        current_price=amount,
        bid_count=1,
        winner_id=payer.id,
    )
    payment = Payment(
        auction_id=auction.id,
        payer_id=payer.id,
        amount=amount,
        status=status,
        expires_at=now + expires_in,
        external_id=external_id,
    )
    db.add(payment)
    await db.flush()
    return auction, payment


def signed(body: dict, *, tamper: bytes | None = None) -> tuple[bytes, dict[str, str]]:
    """PG 가 보내는 것과 똑같이: 본문 바이트를 비밀키로 서명해 헤더에 싣는다."""
    raw = json.dumps(body).encode()
    sig = sign_payload(raw, get_settings().pg_webhook_secret)
    return (tamper or raw), {"X-PG-Signature": sig, "Content-Type": "application/json"}


def webhook_body(payment: Payment, *, event_id: str = "evt_0001", status: str = "paid") -> dict:
    return {
        "event_id": event_id,
        "external_id": payment.external_id,
        "payment_id": str(payment.id),
        "status": status,
        "amount": payment.amount,
    }

async def test_checkout_calls_pg_and_stores_external_id(

    await db_session.refresh(payment)
    assert payment.external_id == "pg_test_0001"

3. app/core/config.py + .env.example — 결제 설정 다섯 개 (수정)

4. app/core/security.py — 서명 두 함수 (수정)

3단계의 해싱·JWT 파일에 HMAC 두 줄을 더합니다. sign_payload 는 바이트를 받습니다 — 문자열이나 dict 가 아니라. 그래야 보내는 쪽과 받는 쪽이 정확히 같은 입력을 씁니다. verify_signature 는 hmac.compare_digest 로 비교합니다. == 는 첫 글자가 다르면 바로 멈추고 열 글자가 맞으면 열 글자만큼 더 걸립니다 — 공격자가 응답 시간으로 서명을 한 글자씩 맞춰 갈 수 있습니다(타이밍 공격). compare_digest 는 길이가 같으면 늘 같은 시간이 걸립니다.

python(수정) app/core/security.py
"""비밀번호 해싱, JWT 발급/검증, 웹훅 서명(9단계). 여기엔 DB 도 Redis 도 없다 — 순수 계산만."""

import hashlib
import hmac
import uuid
from datetime import UTC, datetime, timedelta
from typing import Any, Literal

import jwt
from pwdlib import PasswordHash
from pwdlib.hashers.bcrypt import BcryptHasher

from app.core.config import get_settings

ALGORITHM = "HS256"
TokenType = Literal["access", "refresh"]

_password_hash = PasswordHash((BcryptHasher(),))


def hash_password(raw: str) -> str:
    return _password_hash.hash(raw)


def verify_password(raw: str, hashed: str) -> bool:
    return _password_hash.verify(raw, hashed)


class TokenError(Exception):
    """서명 불일치, 만료, 타입 불일치 등 — 호출자가 401 로 바꾼다."""


def create_token(subject: str, token_type: TokenType, expires_delta: timedelta) -> tuple[str, str]:
    """(토큰 문자열, jti) 를 돌려준다. jti 는 토큰마다 다른 고유 id — refresh 폐기에 쓴다."""
    now = datetime.now(UTC)
    jti = uuid.uuid4().hex
    payload = {
        "sub": subject,
        "type": token_type,
        "jti": jti,
        "iat": now,
        "exp": now + expires_delta,
    }
    token = jwt.encode(payload, get_settings().jwt_secret, algorithm=ALGORITHM)
    return token, jti


def create_access_token(user_id: uuid.UUID) -> str:
    minutes = get_settings().access_token_minutes
    token, _ = create_token(str(user_id), "access", timedelta(minutes=minutes))
    return token


def create_refresh_token(user_id: uuid.UUID) -> tuple[str, str, int]:
    """(토큰, jti, 만료까지 남은 초). 마지막 값은 Redis TTL 로 그대로 쓴다."""
    ttl = timedelta(days=get_settings().refresh_token_days)
    token, jti = create_token(str(user_id), "refresh", ttl)
    return token, jti, int(ttl.total_seconds())


def decode_token(token: str, expected_type: TokenType) -> dict[str, Any]:
    try:
        payload = jwt.decode(token, get_settings().jwt_secret, algorithms=[ALGORITHM])
    except jwt.PyJWTError as exc:  # 만료(ExpiredSignatureError) 도 여기에 포함된다
        raise TokenError(str(exc)) from exc
    if payload.get("type") != expected_type:
        raise TokenError(f"{expected_type} 토큰이 아닙니다")
    return payload


# ── 웹훅 서명 (9단계) ─────────────────────────────────────────────────


def sign_payload(body: bytes, secret: str) -> str:
    """본문 *바이트 그대로* 를 비밀키로 HMAC-SHA256. 보내는 쪽(PG)과 받는 쪽(우리)이 같은 계산.
    한 바이트라도 다르면(공백 하나, 키 순서 하나) 값이 완전히 달라진다 — 그래서 파싱 전
    원문 바이트로 검사한다."""
    return hmac.new(secret.encode(), body, hashlib.sha256).hexdigest()


def verify_signature(body: bytes, signature: str | None, secret: str) -> bool:
    """compare_digest: 앞에서부터 비교하다 첫 불일치에서 멈추지 않는다 → '몇 글자까지 맞았는지' 가
    응답 시간으로 새지 않는다(타이밍 공격 방지). == 로 비교하면 안 되는 이유."""
    if not signature:
        return False
    return hmac.compare_digest(sign_payload(body, secret), signature)

import uuid
from typing import Annotated

import httpx
from fastapi import Depends
from fastapi.security import HTTPAuthorizationCredentials, HTTPBearer
from redis.asyncio import Redis
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.db import get_db
from app.core.exceptions import UnauthorizedError

from app.core.redis import get_redis

from app.repositories import user as user_repo

DbSession = Annotated[AsyncSession, Depends(get_db)]
RedisClient = Annotated[Redis, Depends(get_redis)]
HttpClient = Annotated[httpx.AsyncClient, Depends(get_http)]  # 바깥(PG)으로 나가는 요청용

# auto_error=False: 헤더가 없어도 FastAPI 가 403 을 내지 않고 None 을 준다 → 우리가 401 로 통일
_bearer = HTTPBearer(auto_error=False)


async def get_current_user(
    credentials: Annotated[HTTPAuthorizationCredentials | None, Depends(_bearer)],
    db: DbSession,
) -> User:
    if credentials is None:
        raise UnauthorizedError("인증 토큰이 필요합니다")
    try:
        payload = decode_token(credentials.credentials, expected_type="access")
    except TokenError as exc:
        raise UnauthorizedError("토큰이 유효하지 않거나 만료됐습니다") from exc
    user = await user_repo.get_by_id(db, uuid.UUID(payload["sub"]))
    if user is None:
        raise UnauthorizedError("존재하지 않는 사용자입니다")
    return user


CurrentUser = Annotated[User, Depends(get_current_user)]

async def app_error_handler(request: Request, exc: AppError) -> JSONResponse:
    headers = {"WWW-Authenticate": "Bearer"} if exc.status_code == 401 else None
    return JSONResponse(
        status_code=exc.status_code,
        content={"code": exc.code, "message": exc.message},
        headers=headers,
    )


async def validation_error_handler(request: Request, exc: RequestValidationError) -> JSONResponse:
    """Pydantic 검증 실패(422). 기본 응답은 {"detail": [...]} 인데, 우리 껍데기로 감싸고
    필드별 상세는 "errors" 에 넣는다 — 프론트가 입력칸 옆에 메시지를 보여줄 때 쓴다.
    (loc / msg / type 만 남긴다. 기본 목록에 딸려 오는 input·ctx 는 사용자 입력을 되돌려 주거나
    파이썬 객체라 JSON 으로 못 만드는 경우가 있어 뺀다.)"""
    errors = [{"loc": e["loc"], "msg": e["msg"], "type": e["type"]} for e in exc.errors()]
    return JSONResponse(
        status_code=status.HTTP_422_UNPROCESSABLE_CONTENT,
        content={
            "code": "VALIDATION_ERROR",
            "message": "요청 값이 올바르지 않습니다",
            "errors": errors,
        },
    )


async def http_exception_handler(request: Request, exc: StarletteHTTPException) -> JSONResponse:
    """라우트 없음(404)·허용되지 않은 메서드(405) 같은 프레임워크 기본 에러도 같은 껍데기로."""
    codes = {404: "NOT_FOUND", 405: "METHOD_NOT_ALLOWED"}
    code = codes.get(exc.status_code, f"HTTP_{exc.status_code}")
    return JSONResponse(
        status_code=exc.status_code,
        content={"code": code, "message": exc.detail},
        headers=exc.headers,
    )


def register_exception_handlers(app: FastAPI) -> None:
    app.add_exception_handler(AppError, app_error_handler)  # type: ignore[arg-type]
    app.add_exception_handler(RequestValidationError, validation_error_handler)  # type: ignore[arg-type]
    app.add_exception_handler(StarletteHTTPException, http_exception_handler)  # type: ignore[arg-type]
