async def test_access_log_is_json_with_request_id(client: AsyncClient, caplog):
    with caplog.at_level(logging.INFO, logger="app.access"):
        await client.get("/health", headers={"X-Request-ID": "req-log-1"})

    record = next(r for r in caplog.records if r.name == "app.access")
    line = json.loads(JsonFormatter().format(record))
    assert line["request_id"] == "req-log-1"
    assert (line["method"], line["path"], line["status"]) == ("GET", "/health", 200)
    assert isinstance(line["duration_ms"], int | float)
    assert line["level"] == "INFO"


async def test_slow_query_is_logged(db_session: AsyncSession, caplog):
    remove = install_slow_query_logging(engine, threshold_ms=0)  # 모든 쿼리를 '느린' 것으로
    try:
        with caplog.at_level(logging.WARNING, logger="app.db"):
            await db_session.execute(text("SELECT pg_sleep(0.01)"))
    finally:
        remove()

    # 임계값 0 이면 SAVEPOINT 같은 문장도 전부 찍힌다 → pg_sleep 이 든 것만 고른다
    record = next(r for r in caplog.records if r.name == "app.db" and "pg_sleep" in r.statement)
    line = json.loads(JsonFormatter().format(record))
    assert line["msg"] == "slow query"
    assert "pg_sleep" in line["statement"]
    assert line["duration_ms"] >= 10

async def _ensure_database_exists(database_url: str) -> None:
    """auction_test DB 가 없으면 만든다. CREATE DATABASE 는 트랜잭션 밖에서만 되므로 AUTOCOMMIT."""
    url = make_url(database_url)
    admin = create_async_engine(url.set(database="postgres"), isolation_level="AUTOCOMMIT")
    async with admin.connect() as conn:
        exists = await conn.scalar(
            text("SELECT 1 FROM pg_database WHERE datname = :name"), {"name": url.database}
        )
        if not exists:
            await conn.execute(text(f'CREATE DATABASE "{url.database}"'))
    await admin.dispose()

class QueryCounter:
    """실행된 SQL 문장을 모아 두는 작은 상자. N+1 을 '숫자'로 증명할 때 쓴다."""

    def __init__(self) -> None:
        self.statements: list[str] = []

    def record(self, statement: str) -> None:
        self.statements.append(statement)

    def reset(self) -> None:
        self.statements.clear()

    @property
    def selects(self) -> int:
        return sum(1 for s in self.statements if s.lstrip().upper().startswith("SELECT"))


@pytest.fixture
def query_counter() -> Iterator[QueryCounter]:
    """엔진이 커서에 SQL 을 보내기 직전(before_cursor_execute)마다 문장을 기록한다.
    async 엔진도 속은 동기 엔진(engine.sync_engine)이라 거기에 이벤트를 건다."""
    counter = QueryCounter()

    def _before_cursor_execute(conn, cursor, statement, parameters, context, executemany):  # noqa: ANN001
        counter.record(statement)

    event.listen(engine.sync_engine, "before_cursor_execute", _before_cursor_execute)
    yield counter
    event.remove(engine.sync_engine, "before_cursor_execute", _before_cursor_execute)

from typing import Annotated, Literal

from pydantic import field_validator
from pydantic_settings import BaseSettings, NoDecode, SettingsConfigDict


class Settings(BaseSettings):

    debug: bool = False

    access_token_minutes: int = 15

    snipe_extend_seconds: int = 30  # 마감을 이만큼 늦춘다 (스나이핑 방지)
    bid_rate_limit_per_second: int = 5  # 한 사용자가 1초에 보낼 수 있는 입찰 수

    # 마감 워커 (8단계) — 루프 주기·한 틱에 집는 건수·낙찰 후 결제 기한
    closer_interval_seconds: float = 1.0  # 워커가 시계를 보는 주기 (테스트에선 0.05 로 줄인다)
    closer_batch_size: int = 100  # 한 틱에 처리할 최대 건수 (한 틱이 너무 길어지지 않게)
    payment_deadline_hours: int = 24  # 낙찰 뒤 이 시간 안에 결제하지 않으면 expired

    # 결제 연동 (9단계) — PG 주소, 웹훅 서명 비밀키(PG 와 공유), 바깥 호출 타임아웃·재시도,
    # 그리고 PG 가 웹훅을 보낼 우리 쪽 공개 주소

    pg_webhook_secret: str = "dev-only-pg-webhook-secret-change-me-32-bytes-min"
    pg_timeout_seconds: float = 3.0  # PG 응답을 이보다 오래 기다리지 않는다
    pg_retries: int = 2  # 타임아웃·5xx 면 이만큼 더 시도 (총 3회)
    public_base_url: str = "http://localhost:8000"  # callback_url 의 앞부분

    # 운영 모양 (10단계) — 캐시·로그·CORS·부하 데모
    list_cache_ttl_seconds: int = 5  # GET /auctions 응답을 Redis 에 붙여 두는 시간(초)
    slow_query_ms: int = 200  # 이보다 오래 걸린 SQL 은 WARNING 으로 남긴다
    # CORS_ORIGINS=http://localhost:3000,https://app.example  (콤마로 나열. 비어 있으면 CORS 끔)
    cors_origins: Annotated[list[str], NoDecode] = []
    # 부하 데모용: GET /auctions/{id} 에 일부러 넣은 블로킹 호출을 켠다
    demo_blocking_sleep: bool = False

    @field_validator("cors_origins", mode="before")

    def split_origins(cls, value: object) -> object:

        if isinstance(value, str):
            return [origin.strip() for origin in value.split(",") if origin.strip()]
        return value

    return Settings()

함수 다섯 개, 전부 몇 줄입니다. 핵심은 list_cache_key 가 파라미터를 이름순으로 정렬하고 None 을 빼는 것 — ?limit=20&sort=newest 와 ?sort=newest&limit=20 은 같은 질문이니 같은 키여야 합니다. 그리고 bump_version 은 INCR 하나 — 원자적이라 워커 여러 개가 동시에 불러도 안전합니다.

"""GET /auctions 캐시 — '버전 키' 방식.

무효화를 "키 지우기" 가 아니라 "버전 올리기" 로 한다. 키에 버전이 박혀 있어서
    auction:list:v7:limit=20,sort=ending_soon
경매가 하나라도 바뀌면 INCR 한 번으로 이전 버전의 키가 전부 자연스럽게 버려진다 —
어떤 키가 있는지 찾아다니며(SCAN) 지울 필요가 없다. 옛 키는 TTL 이 알아서 치운다.
"""

import json
from typing import Any

from redis.asyncio import Redis

from app.schemas.auction import AuctionListParams

VERSION_KEY = "auction:list:version"
KEY_PREFIX = "auction:list"


async def current_version(redis: Redis) -> int:
    value = await redis.get(VERSION_KEY)
    return int(value) if value else 0


async def bump_version(redis: Redis) -> int:
    """목록에 보이는 무언가가 바뀌었다 — 생성·수정·취소·입찰·활성화·마감 뒤에 부른다."""
    return await redis.incr(VERSION_KEY)


def list_cache_key(params: AuctionListParams, version: int) -> str:
    """같은 질문이면 같은 키. 필드를 이름순으로 정렬하고 None 은 뺀다 —
    ?limit=20&sort=newest 와 ?sort=newest&limit=20 이 같은 키가 되게."""
    fields = params.model_dump(mode="json", exclude_none=True)
    query = ",".join(f"{k}={fields[k]}" for k in sorted(fields))
    return f"{KEY_PREFIX}:v{version}:{query}"


async def get_json(redis: Redis, key: str) -> Any | None:
    raw = await redis.get(key)
    return json.loads(raw) if raw else None


async def set_json(redis: Redis, key: str, value: Any, ttl_seconds: int) -> None:
    await redis.set(key, json.dumps(value, ensure_ascii=False), ex=ttl_seconds)

"""경매 규칙. HTTP 를 모른다 — 도메인 예외(AppError 계열) 만 던진다.

상태 머신 (한 곳에서만 바꾼다)
    scheduled ──(시작 시각 도달, 8단계 워커)──▶ active ──(마감, 8단계)──▶ ended
        └────────────(판매자 취소, 입찰 0건)────────┴──▶ cancelled
"""

import uuid
from datetime import UTC, datetime

from redis.asyncio import Redis
from sqlalchemy.ext.asyncio import AsyncSession

from app.core import cache
from app.core.config import get_settings
from app.core.exceptions import (
    AuctionClosedError,
    AuctionHasBidsError,

    InvalidPeriodError,

OPEN_STATUSES = {AuctionStatus.SCHEDULED, AuctionStatus.ACTIVE}


def _now() -> datetime:
    return datetime.now(UTC)


async def create(db: AsyncSession, redis: Redis, seller: User, data: AuctionCreate) -> Auction:
    now = _now()
    starts_at = data.starts_at or now
    auction = Auction(
        seller_id=seller.id,
        title=data.title,
        description=data.description,
        category=data.category,
        starting_price=data.starting_price,
        current_price=data.starting_price,  # 입찰 전 현재가 = 시작가
        min_increment=data.min_increment,
        starts_at=starts_at,
        ends_at=data.ends_at,
        status=AuctionStatus.ACTIVE if starts_at <= now else AuctionStatus.SCHEDULED,
    )
    await auction_repo.add(db, auction)
    await db.commit()
    await db.refresh(auction)  # DB 가 채운 created_at / updated_at 을 읽어 온다
    await cache.bump_version(redis)  # 목록이 달라졌다 — 캐시 버전을 올린다 (커밋 뒤에)
    return auction


async def list_auctions(db: AsyncSession, redis: Redis, params: AuctionListParams) -> AuctionPage:
    """한 페이지 + 다음 커서. 캐시 → (미스면) 커서 해석 · 쿼리 · 직렬화 · 캐시 저장.

    캐시 키에 '현재 버전' 이 박힌다. 경매가 바뀔 때마다 bump_version() 이 버전을 올리므로
    옛 키는 다시는 조회되지 않고 TTL 로 사라진다. 같은 버전·같은 파라미터면 DB 를 안 건드린다.
    """
    version = await cache.current_version(redis)
    key = cache.list_cache_key(params, version)
    cached = await cache.get_json(redis, key)
    if cached is not None:
        return AuctionPage.model_validate(cached)

    after = decode_cursor(params.cursor, params.sort) if params.cursor else None
    rows = await auction_repo.list_page(
        db,
        status=params.status,
        category=params.category,
        min_price=params.min_price,
        max_price=params.max_price,
        sort=params.sort,
        limit=params.limit,
        after=after,
    )
    items = rows[: params.limit]
    next_cursor = None
    if len(rows) > params.limit:  # limit+1 번째가 있다 = 다음 페이지가 있다
        last = items[-1]
        sort_key = last.ends_at if params.sort == "ending_soon" else last.created_at
        next_cursor = encode_cursor(params.sort, sort_key, last.id)

    page = AuctionPage(items=items, next_cursor=next_cursor)
    ttl = get_settings().list_cache_ttl_seconds
    await cache.set_json(redis, key, page.model_dump(mode="json"), ttl)
    return page


async def get(db: AsyncSession, auction_id: uuid.UUID) -> Auction:
    auction = await auction_repo.get(db, auction_id)
    if auction is None:
        raise AuctionNotFoundError()
    return auction


def _ensure_seller(auction: Auction, user: User) -> None:
    if auction.seller_id != user.id:
        raise NotAuctionSellerError()


def _ensure_open_and_unbid(auction: Auction) -> None:
    """수정·취소 공통 규칙: 아직 열려 있고(scheduled/active), 입찰이 하나도 없어야 한다."""
    if auction.status not in OPEN_STATUSES:
        raise AuctionClosedError()
    if auction.bid_count > 0:
        raise AuctionHasBidsError()


async def update(
    db: AsyncSession, redis: Redis, user: User, auction_id: uuid.UUID, data: AuctionUpdate
) -> Auction:
    auction = await get(db, auction_id)
    _ensure_seller(auction, user)
    _ensure_open_and_unbid(auction)

    changes = data.model_dump(exclude_unset=True)  # 요청에 실제로 담긴 필드만
    if "ends_at" in changes and changes["ends_at"] <= auction.starts_at:
        raise InvalidPeriodError()
    for field, value in changes.items():
        setattr(auction, field, value)
    auction.version += 1  # 6단계 낙관적 락의 재료 — 바뀔 때마다 1 증가

    await db.commit()
    await db.refresh(auction)
    await cache.bump_version(redis)
    return auction


async def cancel(db: AsyncSession, redis: Redis, user: User, auction_id: uuid.UUID) -> Auction:
    auction = await get(db, auction_id)
    _ensure_seller(auction, user)
    _ensure_open_and_unbid(auction)

    auction.status = AuctionStatus.CANCELLED
    auction.version += 1
    await db.commit()
    await db.refresh(auction)
    await cache.bump_version(redis)
    return auction

"""입찰 규칙. HTTP 를 모른다 — 도메인 예외만 던진다.

핵심은 순서다: (rate limit) → **경매 행 잠금** → 규칙 검사 → 갱신 → 커밋.
잠금부터 커밋까지가 한 트랜잭션이라, 그 사이에 다른 입찰은 줄을 서서 기다린다.
그래서 '현재가 + 최소 단위' 를 계산할 때 본 값이 커밋하는 순간까지 그대로다.
"""

import uuid
from datetime import UTC, datetime, timedelta

from redis.asyncio import Redis
from sqlalchemy.ext.asyncio import AsyncSession

from app.core import cache, events
from app.core.config import get_settings
from app.core.exceptions import (
    AuctionNotActiveError,
    AuctionNotFoundError,
    BidRateLimitedError,
    BidTooLowError,
    OwnAuctionBidError,
)
from app.models import Auction, AuctionStatus, Bid, User
from app.repositories import auction as auction_repo
from app.repositories import bid as bid_repo


def _now() -> datetime:
    """시계를 함수로 빼 두면 테스트에서 멈추거나 돌릴 수 있다."""
    return datetime.now(UTC)


async def _check_rate_limit(redis: Redis, user_id: uuid.UUID, now: datetime) -> None:
    """사용자별 '이번 1초' 카운터. INCR 는 원자적이라 동시에 들어와도 정확히 센다."""
    key = f"bid:rl:{user_id}:{int(now.timestamp())}"
    count = await redis.incr(key)
    if count == 1:
        await redis.expire(key, 2)  # 1초짜리 키가 영원히 남지 않도록
    if count > get_settings().bid_rate_limit_per_second:
        raise BidRateLimitedError()


def minimum_bid(auction: Auction) -> int:
    """첫 입찰은 시작가 이상, 그 다음부터는 현재가 + 최소 단위 이상."""
    if auction.bid_count == 0:
        return auction.starting_price
    return auction.current_price + auction.min_increment


async def place_bid(
    db: AsyncSession, redis: Redis, bidder: User, auction_id: uuid.UUID, amount: int
) -> tuple[Bid, Auction]:
    settings = get_settings()
    now = _now()

    await _check_rate_limit(redis, bidder.id, now)  # 락을 잡기 *전에* — 스팸이 락을 쥐고 있지 않게

    auction = await auction_repo.get_for_update(db, auction_id)  # ← 여기서부터 줄을 선다
    if auction is None:
        raise AuctionNotFoundError()
    if auction.status != AuctionStatus.ACTIVE or now >= auction.ends_at:
        raise AuctionNotActiveError()
    if auction.seller_id == bidder.id:
        raise OwnAuctionBidError()
    minimum = minimum_bid(auction)
    if amount < minimum:
        raise BidTooLowError(
            f"최소 입찰액은 {minimum:,}원입니다 (현재가 {auction.current_price:,}원)"
        )

    # 규칙 4 — 마감 직전 입찰이면 마감을 늦춘다 (스나이핑 방지)
    extended = False
    if auction.ends_at - now < timedelta(seconds=settings.snipe_window_seconds):
        auction.ends_at = auction.ends_at + timedelta(seconds=settings.snipe_extend_seconds)
        extended = True

    auction.current_price = amount
    auction.bid_count += 1
    auction.version += 1
    bid = Bid(auction_id=auction.id, bidder_id=bidder.id, amount=amount)
    await bid_repo.add(db, bid)

    await db.commit()  # ← 여기서 락이 풀리고, 줄 서 있던 다음 입찰이 '바뀐 현재가' 를 본다
    await db.refresh(bid)
    await db.refresh(auction)
    await cache.bump_version(redis)  # 목록의 current_price·bid_count 가 바뀌었다

    # 커밋 *뒤에* 방송한다. 커밋 전에 방송하면 구독자가 "아직 DB 에 없는 입찰" 을 보게 되고,
    # 커밋이 실패하면 없던 일이 방송된 채로 남는다.
    await events.publish(
        redis,
        events.AuctionEvent(
            type="bid_placed",
            auction_id=auction.id,
            payload={
                "amount": bid.amount,
                "bidder_id": bid.bidder_id,
                "current_price": auction.current_price,
                "bid_count": auction.bid_count,
                "ends_at": auction.ends_at,
                "version": auction.version,
                "extended": extended,
            },
        ),
    )
    return bid, auction


async def list_bids(db: AsyncSession, auction_id: uuid.UUID) -> list[Bid]:
    if await auction_repo.get(db, auction_id) is None:
        raise AuctionNotFoundError()
    return await bid_repo.list_by_auction(db, auction_id)

"""시간이 흐르면 저절로 일어나야 하는 일 세 가지. HTTP 도, 루프도, 시그널도 모른다.

  activate_due   : 시작 시각이 지난 scheduled 경매 → active
  close_due      : 마감 시각이 지난 active 경매 → ended (+ 낙찰자, Payment, auction_closed 이벤트)
  expire_payments: 결제 기한이 지난 pending 결제 → expired

세 함수 모두 `now` 를 인자로 받는다. 워커는 진짜 시계를 넣고, 테스트는 원하는 시각을 넣는다.
누가 부르든(워커 1개든 2개든) 결과가 같아야 한다 — 그게 이 파일의 유일한 어려움이다.
"""

import logging
import uuid
from dataclasses import dataclass
from datetime import datetime, timedelta

from redis.asyncio import Redis
from sqlalchemy import update
from sqlalchemy.ext.asyncio import AsyncSession

from app.core import cache, events
from app.core.config import get_settings
from app.models import Auction, AuctionStatus, Payment, PaymentStatus
from app.repositories import auction as auction_repo
from app.repositories import bid as bid_repo
from app.repositories import payment as payment_repo

logger = logging.getLogger(__name__)


@dataclass(frozen=True)
class ClosedResult:
    auction_id: uuid.UUID
    winner_id: uuid.UUID | None  # None = 유찰 (입찰 0건)
    final_price: int | None
    bid_count: int
    payment_id: uuid.UUID | None


async def activate_due(db: AsyncSession, now: datetime, *, limit: int | None = None) -> int:
    """UPDATE 한 문장. UPDATE 는 스스로 행을 잠그므로 워커가 둘이어도 안전하다 —
    둘째 워커는 첫째의 커밋을 기다렸다가 다시 조건을 보고(이미 active) 0 행을 바꾼다."""
    limit = limit or get_settings().closer_batch_size
    ids = await auction_repo.list_due_scheduled_ids(db, now, limit=limit)
    if not ids:
        return 0
    stmt = (
        update(Auction)
        .where(Auction.id.in_(ids), Auction.status == AuctionStatus.SCHEDULED)  # 조건을 한 번 더
        .values(status=AuctionStatus.ACTIVE, version=Auction.version + 1)
        .returning(Auction.id)
    )
    activated = list((await db.scalars(stmt)).all())
    await db.commit()
    if activated:
        logger.info("activated %d auction(s): %s", len(activated), [str(i)[:8] for i in activated])
    return len(activated)


async def close_due(
    db: AsyncSession, redis: Redis, now: datetime, *, limit: int | None = None
) -> list[ClosedResult]:
    """후보 목록은 락 없이 값싸게 뽑고, 실제 처리는 **한 건씩** 잠그고 커밋한다.

    왜 한 건씩인가: 100건을 한 트랜잭션에 묶으면 (1) 100번째에서 실패하면 99건이 같이 롤백되고,
    (2) 그동안 100개 행이 잠겨 입찰 요청들이 줄을 선다. 한 건씩이면 실패도 잠금도 한 건뿐이다.
    왜 skip_locked 인가: 다른 워커가 이미 잡은 행이면 기다릴 이유가 없다 — 그쪽이 끝낼 것이다.
    왜 재검사인가: 후보를 뽑은 뒤 잠그기까지 사이에 누가 이미 닫았을 수 있다(그 워커가 커밋해서
    락을 풀면 우리는 status=ended 인 행을 잠그게 된다). 잠근 *뒤에* 본 값만 믿는다.
    """
    settings = get_settings()
    limit = limit or settings.closer_batch_size
    results: list[ClosedResult] = []

    for auction_id in await auction_repo.list_due_active_ids(db, now, limit=limit):
        auction = await auction_repo.get_for_update(db, auction_id, skip_locked=True)
        if auction is None:  # 다른 워커가 잠갔거나(skip) 사라졌다 → 내 몫이 아니다
            continue
        if auction.status != AuctionStatus.ACTIVE or auction.ends_at > now:  # 잠근 뒤 재검사
            await db.rollback()  # 잠금만 풀고 다음
            continue

        top = await bid_repo.top_bid(db, auction.id)
        auction.status = AuctionStatus.ENDED
        auction.winner_id = top.bidder_id if top else None
        auction.version += 1

        payment: Payment | None = None
        if top is not None:
            payment = await payment_repo.add(
                db,
                Payment(
                    auction_id=auction.id,
                    payer_id=top.bidder_id,
                    amount=auction.current_price,
                    status=PaymentStatus.PENDING,
                    expires_at=now + timedelta(hours=settings.payment_deadline_hours),
                ),
            )
        await db.commit()  # ← 여기서 이 경매의 락이 풀린다. 아래는 커밋된 사실만 방송한다
        await cache.bump_version(redis)  # 목록의 status 가 ended 로 바뀌었다

        result = ClosedResult(
            auction_id=auction.id,
            winner_id=auction.winner_id,
            final_price=auction.current_price if top else None,
            bid_count=auction.bid_count,
            payment_id=payment.id if payment else None,
        )
        await events.publish(
            redis,
            events.AuctionEvent(
                type="auction_closed",
                auction_id=auction.id,
                payload={
                    "winner_id": result.winner_id,
                    "final_price": result.final_price,
                    "bid_count": result.bid_count,
                    "payment_id": result.payment_id,
                    "payment_expires_at": payment.expires_at if payment else None,
                    "version": auction.version,
                },
            ),
        )
        if top is not None:
            # 진짜라면 여기서 메일/푸시 큐에 넣는다. 이 프로젝트에서는 로그 한 줄이 '알림' 이다.
            logger.info(
                "[notify] user=%s 낙찰! auction=%s %s원 — %s 까지 결제해 주세요",
                str(top.bidder_id)[:8],
                str(auction.id)[:8],
                f"{auction.current_price:,}",
                payment.expires_at.isoformat(timespec="minutes") if payment else "",
            )
        else:
            logger.info("[notify] auction=%s 유찰 (입찰 0건)", str(auction.id)[:8])
        results.append(result)
    return results


async def expire_payments(db: AsyncSession, now: datetime) -> int:
    """기한 지난 pending → expired. activate_due 와 같은 'UPDATE 한 문장' 패턴."""
    stmt = (
        update(Payment)
        .where(Payment.status == PaymentStatus.PENDING, Payment.expires_at <= now)
        .values(status=PaymentStatus.EXPIRED)
        .returning(Payment.id)
    )
    expired = list((await db.scalars(stmt)).all())
    await db.commit()
    if expired:
        logger.info("expired %d payment(s): %s", len(expired), [str(i)[:8] for i in expired])
    return len(expired)

워커는 두 가지가 바뀝니다. 활성화 뒤 버전 올리기, 그리고 logging.basicConfig 대신 API 와 같은 setup_logging() — 한 수집기에서 두 프로세스의 로그가 같은 모양으로 보여야 합니다(실행·확인의 워커 로그가 JSON 인 이유).

python(수정) app/workers/closer.py
"""마감 워커 — API 서버와 **별개의 프로세스**.  실행:  uv run python -m app.workers.closer

하는 일은 단순하다: interval 초마다 시계를 보고 services/closer 의 세 함수를 차례로 부른다.
규칙은 전부 서비스에 있고, 이 파일은 "언제 부르나 · 어떻게 멈추나 · 무엇을 남기나" 만 안다.

왜 API 안의 BackgroundTasks 나 asyncio.create_task 가 아닌가:
  - 요청이 없어도 돌아야 한다 (새벽 3시에 마감되는 경매도 제때 닫혀야 한다)
  - uvicorn --workers 4 면 태스크도 4개가 되고, 서버를 재시작하면 사라진다
  - 별도 프로세스면 API 와 독립적으로 죽이고 살리고 늘릴 수 있다 (10단계 compose 의 `worker` 서비스)
"""

import asyncio
import logging
import signal
from dataclasses import dataclass, field
from datetime import UTC, datetime

from redis.asyncio import Redis

from app.core import cache
from app.core.config import get_settings
from app.core.db import SessionLocal, engine
from app.core.logging import install_slow_query_logging, setup_logging
from app.services import closer
from app.services.closer import ClosedResult

logger = logging.getLogger("worker.closer")


@dataclass
class TickResult:
    activated: int = 0
    closed: list[ClosedResult] = field(default_factory=list)
    expired: int = 0

    @property
    def empty(self) -> bool:
        return self.activated == 0 and not self.closed and self.expired == 0


async def tick(redis: Redis, now: datetime | None = None) -> TickResult:
    """한 번의 '시계 보기'. 함수마다 세션을 새로 여는 이유: 하나가 실패해도 다른 둘은 독립적으로
    커밋되고, 세션(커넥션)을 틱 사이에 쥐고 있지 않기 위해서다."""
    now = now or datetime.now(UTC)
    result = TickResult()
    async with SessionLocal() as db:
        result.activated = await closer.activate_due(db, now)
    if result.activated:
        await cache.bump_version(redis)  # scheduled → active: 목록의 status 가 바뀌었다
    async with SessionLocal() as db:
        result.closed = await closer.close_due(db, redis, now)
    async with SessionLocal() as db:
        result.expired = await closer.expire_payments(db, now)
    return result


async def run(stop: asyncio.Event, redis: Redis, interval: float) -> None:
    """stop 이 set 될 때까지 interval 마다 tick. 한 틱이 실패해도 루프는 죽지 않는다(로그만)."""
    logger.info("closer started (interval=%ss)", interval)
    while not stop.is_set():
        try:
            result = await tick(redis)
            if not result.empty:
                logger.info(
                    "tick: activated=%d closed=%d expired=%d",
                    result.activated,
                    len(result.closed),
                    result.expired,
                )
        except Exception:  # noqa: BLE001 — 워커는 한 틱의 실패로 죽으면 안 된다
            logger.exception("tick failed")
        try:
            # sleep 하면 종료 신호를 interval 만큼 늦게 받는다. stop.wait() 를 기다리면 즉시 깬다.
            await asyncio.wait_for(stop.wait(), timeout=interval)
        except TimeoutError:
            pass
    logger.info("closer stopped")


async def main() -> None:
    settings = get_settings()
    setup_logging()  # API 와 같은 JSON 로그 — 한 수집기에서 두 프로세스를 같은 모양으로 본다
    install_slow_query_logging(engine, settings.slow_query_ms)
    redis = Redis.from_url(settings.redis_url, decode_responses=True)  # 워커 전용 클라이언트
    stop = asyncio.Event()
    loop = asyncio.get_running_loop()
    for sig in (signal.SIGINT, signal.SIGTERM):  # Ctrl-C, docker stop → 조용히 마무리
        loop.add_signal_handler(sig, stop.set)
    try:
        await run(stop, redis, settings.closer_interval_seconds)
    finally:
        await redis.aclose()
        await engine.dispose()


if __name__ == "__main__":
    asyncio.run(main())
