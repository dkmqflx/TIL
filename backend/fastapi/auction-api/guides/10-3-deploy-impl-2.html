7. app/core/logging.py — JSON 로그 · request-id · 느린 쿼리

세 부분입니다. JsonFormatter 는 LogRecord 를 JSON 한 줄로 — 기본 속성 외에 extra={…} 로 들어온 우리 필드(method·status·duration_ms·statement)를 전부 싣고, request_id 는 레코드에 있으면 그것, 없으면 contextvar 에서 꺼냅니다. request_id_middleware 는 헤더가 있으면 재사용·없으면 발급하고, contextvar 에 넣고(set), 끝나면 되돌리고(reset), 응답 헤더로 돌려주고, access 로그 한 줄을 남깁니다 — uvicorn 의 access 로그는 끕니다(같은 내용을 request_id 없이 두 번 찍을 이유가 없습니다). install_slow_query_logging 은 5단계 query_counter 와 같은 SQLAlchemy 커서 이벤트로 SQL 하나의 소요 시간을 재서 임계값을 넘으면 WARNING 을 남깁니다 — 되돌리는 함수를 돌려줘서 테스트가 임계값 0 짜리를 잠깐 달았다 뗄 수 있습니다.

pythonapp/core/logging.py
"""JSON 로그 · request-id · 느린 쿼리 경고.

로그 한 줄 = JSON 한 개. 사람이 아니라 수집기(Loki·CloudWatch·Datadog)가 읽는다고 생각하고 쓴다.
request_id 는 contextvar 에 둔다 — 같은 요청을 처리하는 동안 어디서 찍힌 로그든 자동으로 따라붙는다.
"""

import contextvars
import json
import logging
import sys
import time
import uuid
from collections.abc import Awaitable, Callable
from datetime import UTC, datetime

from fastapi import Request, Response
from sqlalchemy import event
from sqlalchemy.ext.asyncio import AsyncEngine

REQUEST_ID_HEADER = "X-Request-ID"
request_id_var: contextvars.ContextVar[str] = contextvars.ContextVar("request_id", default="-")

# LogRecord 가 원래 갖고 있는 속성들 — 이 밖의 것은 extra={} 로 들어온 우리 필드다
_STANDARD = set(logging.LogRecord("x", 0, "", 0, "", (), None).__dict__) | {
    "message",
    "asctime",
    "taskName",
}


class JsonFormatter(logging.Formatter):
    def format(self, record: logging.LogRecord) -> str:
        line: dict[str, object] = {
            "ts": datetime.fromtimestamp(record.created, UTC).isoformat(timespec="milliseconds"),
            "level": record.levelname,
            "logger": record.name,
            "msg": record.getMessage(),
            "request_id": getattr(record, "request_id", None) or request_id_var.get(),
        }
        for key, value in record.__dict__.items():
            if key not in _STANDARD and key != "request_id":
                line[key] = value
        if record.exc_info:
            line["exc"] = self.formatException(record.exc_info)
        return json.dumps(line, ensure_ascii=False, default=str)


def setup_logging(level: str = "INFO") -> None:
    """루트 로거 하나에 JSON 핸들러 하나. uvicorn 의 로그도 여기로 흘려보내고,
    uvicorn 의 access 로그는 끈다 — 우리 미들웨어가 request_id 와 소요 시간을 붙여 대신 남긴다."""
    handler = logging.StreamHandler(sys.stdout)
    handler.setFormatter(JsonFormatter())
    root = logging.getLogger()
    root.handlers = [handler]
    root.setLevel(level)
    for name in ("uvicorn", "uvicorn.error"):
        uv = logging.getLogger(name)
        uv.handlers = []
        uv.propagate = True
    access = logging.getLogger("uvicorn.access")
    access.handlers = []
    access.propagate = False


access_logger = logging.getLogger("app.access")
db_logger = logging.getLogger("app.db")


async def request_id_middleware(
    request: Request, call_next: Callable[[Request], Awaitable[Response]]
) -> Response:
    """들어온 X-Request-ID 가 있으면 그대로(게이트웨이·프론트가 붙인 것), 없으면 발급.
    응답 헤더로 돌려주고, 요청 한 줄을 JSON 으로 남긴다."""
    request_id = request.headers.get(REQUEST_ID_HEADER) or str(uuid.uuid4())
    token = request_id_var.set(request_id)
    started = time.perf_counter()
    status = 500  # 처리 중 예외로 끝나면 500 으로 기록된다
    try:
        response = await call_next(request)
        status = response.status_code
    finally:
        access_logger.info(
            "request",
            extra={
                "request_id": request_id,
                "method": request.method,
                "path": request.url.path,
                "status": status,
                "duration_ms": round((time.perf_counter() - started) * 1000, 1),
            },
        )
        request_id_var.reset(token)
    response.headers[REQUEST_ID_HEADER] = request_id
    return response


def install_slow_query_logging(engine: AsyncEngine, threshold_ms: int) -> Callable[[], None]:
    """커서 실행 전후 이벤트로 SQL 하나의 소요 시간을 재고, 임계값을 넘으면 WARNING.
    async 엔진도 속은 동기 엔진(engine.sync_engine)이라 거기에 건다. 되돌리는 함수를 돌려준다."""
    sync_engine = engine.sync_engine
    stack_key = f"slow_query_started:{id(sync_engine)}:{threshold_ms}"

    def before(conn, cursor, statement, parameters, context, executemany):  # noqa: ANN001
        conn.info.setdefault(stack_key, []).append(time.perf_counter())

    def after(conn, cursor, statement, parameters, context, executemany):  # noqa: ANN001
        started = conn.info.get(stack_key)
        if not started:
            return
        elapsed_ms = (time.perf_counter() - started.pop()) * 1000
        if elapsed_ms >= threshold_ms:
            db_logger.warning(
                "slow query",
                extra={
                    "duration_ms": round(elapsed_ms, 1),
                    "statement": " ".join(statement.split())[:300],
                },
            )

    event.listen(sync_engine, "before_cursor_execute", before)
    event.listen(sync_engine, "after_cursor_execute", after)

    def remove() -> None:
        event.remove(sync_engine, "before_cursor_execute", before)
        event.remove(sync_engine, "after_cursor_execute", after)

    return remove
contextvar 가 왜 필요한가. 요청 하나는 미들웨어 → 라우터 → 서비스 → SQL → httpx 를 거치는데, request_id 를 함수 인자로 다 끌고 다닐 순 없습니다. ContextVar 는 "지금 이 코루틴(과 거기서 파생된 태스크·스레드)만의 전역 변수" 입니다. 미들웨어가 한 번 set 하면 그 요청을 처리하는 동안 어디서 get 해도 같은 값이고, 동시에 처리 중인 다른 요청은 자기 값을 봅니다. 실행·확인에서 httpx 라이브러리의 자체 로그에까지 우리 request_id 가 붙는 걸 보게 됩니다 — 우리가 httpx 에 아무것도 안 넘겼는데도요.

8. app/main.py — 로깅 · 느린 쿼리 · CORS · 미들웨어 (수정)

create_app 맨 앞에서 setup_logging() 과 install_slow_query_logging(). CORS 는 허용 출처가 있을 때만 답니다(빈 리스트면 브라우저 교차 출처 요청은 전부 막힌 상태 — 안전한 기본값). 미들웨어는 나중에 add 한 것이 바깥입니다. CORS 를 먼저(안쪽), request-id 를 나중에(바깥) 두면 preflight 응답과 거절(400)에도 X-Request-ID 와 access 로그가 붙습니다. CORS 자체는 7주차·03 에서 다뤘으므로 여기서 다시 설명하지 않습니다(정본 링크).

import asyncio
import pathlib
from collections.abc import AsyncIterator
from contextlib import asynccontextmanager

from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from fastapi.staticfiles import StaticFiles

from app.core.config import get_settings
from app.core.db import engine
from app.core.exceptions import register_exception_handlers
from app.core.http import http_client
from app.core.logging import (
    REQUEST_ID_HEADER,
    install_slow_query_logging,
    request_id_middleware,
    setup_logging,
)
from app.core.redis import redis_client
from app.routers import auctions, auth, bids, health, payments, users, webhooks
from app.ws import router as ws_router
from app.ws.bridge import run_bridge
from app.ws.manager import manager

STATIC_DIR = pathlib.Path(__file__).resolve().parent.parent / "static"


@asynccontextmanager
async def lifespan(app: FastAPI) -> AsyncIterator[None]:
    # 프로세스(워커)마다 Redis 구독 루프 하나 — 이 워커에 붙은 소켓들에게 이벤트를 중계한다
    bridge = asyncio.create_task(run_bridge(redis_client, manager))
    yield
    bridge.cancel()
    try:
        await bridge
    except asyncio.CancelledError:
        pass
    await http_client.aclose()  # 열려 있던 PG 커넥션 정리
    await redis_client.aclose()
    await engine.dispose()


def create_app() -> FastAPI:
    settings = get_settings()
    setup_logging()  # 루트 로거 = JSON 한 줄. uvicorn 로그도 같은 모양으로
    install_slow_query_logging(engine, settings.slow_query_ms)
    app = FastAPI(title="Auction API", debug=settings.debug, lifespan=lifespan)
    register_exception_handlers(app)

    # 미들웨어는 나중에 add 한 것이 바깥이다. CORS 를 먼저(안쪽), request-id 를 나중에(바깥) 둬서
    # preflight 응답까지 request-id 와 access 로그가 붙게 한다.
    if settings.cors_origins:  # 비어 있으면 CORS 를 끈다 — 브라우저 교차 출처 요청이 전부 막힌다
        app.add_middleware(
            CORSMiddleware,
            allow_origins=settings.cors_origins,
            allow_credentials=True,  # 출처를 명시했을 때만 안전하다 ("*" 와는 같이 못 쓴다)
            allow_methods=["*"],
            allow_headers=["*"],
            expose_headers=[REQUEST_ID_HEADER],  # 브라우저 JS 가 X-Request-ID 를 읽을 수 있게
        )
    app.middleware("http")(request_id_middleware)
    app.include_router(health.router)
    app.include_router(auth.router)
    app.include_router(users.router)
    app.include_router(auctions.router)
    app.include_router(bids.router)
    app.include_router(payments.router)
    app.include_router(webhooks.router)
    app.include_router(ws_router.router)
    app.mount("/static", StaticFiles(directory=STATIC_DIR), name="static")
    return app


app = create_app()

9. alembic/env.py — 한 단어 (수정)

이게 없으면 관측성 테스트 두 개가 이유도 모른 채 실패합니다. Alembic 의 env.py 는 fileConfig(alembic.ini) 로 로깅을 설정하는데, 그 함수의 기본값 disable_existing_loggers=True 는 그 시점에 이미 만들어진 모든 로거를 꺼 버립니다. 테스트는 앱을 import 한 뒤(app.access·app.db 로거가 이미 생김) 세션 시작 fixture 에서 alembic upgrade head 를 같은 프로세스로 돌리므로, 우리 로거가 전부 disabled 가 됩니다. 운영에서는 마이그레이션이 별도 프로세스라 문제없지만, 같은 프로세스에서 돌리는 순간 터지는 종류의 함정입니다.

python(수정) alembic/env.py
import asyncio
from logging.config import fileConfig

from alembic import context
from sqlalchemy import pool
from sqlalchemy.engine import Connection
from sqlalchemy.ext.asyncio import async_engine_from_config

from app.core.config import get_settings
from app.models import Base

# Alembic Config 객체 — alembic.ini 의 값에 접근한다
config = context.config

# alembic.ini 의 sqlalchemy.url 대신 앱의 Settings(.env) 에서 접속 주소를 가져온다.
# → 앱과 마이그레이션이 같은 DB 를 보게 되고, 비밀번호가 ini 에 박히지 않는다.
config.set_main_option("sqlalchemy.url", get_settings().database_url)

if config.config_file_name is not None:
    # disable_existing_loggers=False: 기본값(True)은 이 시점에 이미 만들어진 로거(app.access 등)를
    # 전부 꺼 버린다. 테스트처럼 같은 프로세스에서 마이그레이션을 돌리면 앱 로그가 사라진다(10단계).
    fileConfig(config.config_file_name, disable_existing_loggers=False)

# autogenerate 가 "현재 모델" 로 삼는 메타데이터
target_metadata = Base.metadata


def run_migrations_offline() -> None:
    """DB 연결 없이 SQL 만 출력하는 모드 (alembic upgrade head --sql)."""
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
        compare_type=True,
    )
    with context.begin_transaction():
        context.run_migrations()


def do_run_migrations(connection: Connection) -> None:
    context.configure(connection=connection, target_metadata=target_metadata, compare_type=True)
    with context.begin_transaction():
        context.run_migrations()


async def run_async_migrations() -> None:
    """asyncpg 엔진으로 연결한 뒤, 동기 API 인 Alembic 을 run_sync 로 감싸 실행한다."""
    connectable = async_engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )
    async with connectable.connect() as connection:
        await connection.run_sync(do_run_migrations)
    await connectable.dispose()


def run_migrations_online() -> None:
    asyncio.run(run_async_migrations())


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

10. app/services/auth.py — 부하 테스트가 찾아낸 진짜 병목 (수정)

이 수정은 원래 계획에 없었습니다. 실행·확인의 부하 테스트를 처음 돌렸을 때 로그인·회원가입이 유독 느렸고, 재 보니 hash_password 가 202ms, verify_password 가 197ms 였습니다. bcrypt 는 일부러 느린 알고리즘입니다(무차별 대입을 막으려고) — 그 자체는 정상입니다. 문제는 3단계에서 이 동기 함수를 async def 서비스 안에서 그냥 불렀다는 것: 그 200ms 동안 이벤트 루프가 멈춰 다른 모든 요청이 기다립니다. 4주차·02 의 블로킹 함정 그대로입니다. 고치는 법은 한 줄 — await asyncio.to_thread(hash_password, …) 로 스레드에 보냅니다. 3단계 테스트는 전부 그대로 통과합니다.

"""인증 규칙. HTTP 를 모른다 — 도메인 예외(AppError) 만 던진다."""

import asyncio
import uuid

from redis.asyncio import Redis
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.exceptions import EmailAlreadyExistsError, UnauthorizedError
from app.core.security import (
    TokenError,
    create_access_token,
    create_refresh_token,
    decode_token,
    hash_password,
    verify_password,
)
from app.models import User
from app.repositories import user as user_repo
from app.schemas.auth import LoginRequest, SignupRequest, TokenPair


def _refresh_key(jti: str) -> str:
    return f"refresh:{jti}"


async def signup(db: AsyncSession, data: SignupRequest) -> User:
    if await user_repo.get_by_email(db, data.email) is not None:
        raise EmailAlreadyExistsError()
    user = await user_repo.create(
        db,
        email=data.email,
        # bcrypt 는 일부러 느리다(수십~수백 ms). 동기 함수를 그냥 부르면 그 시간 동안
        # 이벤트 루프가 멈춘다 → 스레드로 보낸다 (10단계 부하 테스트가 찾아낸 병목)
        password_hash=await asyncio.to_thread(hash_password, data.password),
        nickname=data.nickname,
    )
    await db.commit()
    await db.refresh(user)  # DB 가 채운 created_at 을 읽어 온다
    return user


async def _issue_token_pair(redis: Redis, user_id: uuid.UUID) -> TokenPair:
    access = create_access_token(user_id)
    refresh, jti, ttl = create_refresh_token(user_id)
    await redis.set(_refresh_key(jti), str(user_id), ex=ttl)  # 살아 있는 refresh 토큰 목록
    return TokenPair(access_token=access, refresh_token=refresh)


async def login(db: AsyncSession, redis: Redis, data: LoginRequest) -> TokenPair:
    user = await user_repo.get_by_email(db, data.email)
    # 이메일이 없는 경우와 비밀번호가 틀린 경우를 구분해 알려주지 않는다 (계정 존재 여부 노출 방지)
    ok = user is not None and await asyncio.to_thread(
        verify_password, data.password, user.password_hash
    )
    if not ok:
        raise UnauthorizedError("이메일 또는 비밀번호가 올바르지 않습니다")
    return await _issue_token_pair(redis, user.id)


async def refresh(redis: Redis, refresh_token: str) -> TokenPair:
    try:
        payload = decode_token(refresh_token, expected_type="refresh")
    except TokenError as exc:
        raise UnauthorizedError("refresh 토큰이 유효하지 않습니다") from exc
    # DEL 은 원자적이다: 같은 토큰으로 동시에 두 번 와도 한 쪽만 1 을 받는다 → 회전(rotation)
    deleted = await redis.delete(_refresh_key(payload["jti"]))
    if not deleted:
        raise UnauthorizedError("이미 사용됐거나 폐기된 refresh 토큰입니다")
    return await _issue_token_pair(redis, uuid.UUID(payload["sub"]))


async def logout(redis: Redis, refresh_token: str) -> None:
    try:
        payload = decode_token(refresh_token, expected_type="refresh")
    except TokenError:
        return  # 이미 못 쓰는 토큰이면 할 일이 없다 — 로그아웃은 멱등
    await redis.delete(_refresh_key(payload["jti"]))

12. loadtest/locustfile.py — 가상 사용자 한 명의 하루

locust 는 "사용자 한 명이 하는 일" 을 파이썬 클래스로 씁니다. on_start 에서 가입·로그인을 한 번 하고, 그 뒤엔 목록 5 : 상세 3 : 입찰 1 의 비율로 반복합니다(@task(5)). 입찰은 대부분 규칙에 걸려 422·429 로 거절되는데, 그건 서버가 일을 제대로 한 것이므로 실패로 세지 않습니다(catch_response). 9주차·03 에서 ab/wrk 로 했던 것의 "시나리오 있는" 버전입니다.

pythonloadtest/locustfile.py
"""부하 테스트 시나리오 — 실행:
    uv run locust -f loadtest/locustfile.py --headless -u 50 -r 10 -t 20s \
        --host http://localhost:8000 --only-summary

가상 사용자 한 명 = 회원가입 → 로그인(한 번) → 목록 5 : 상세 3 : 입찰 1 비율로 반복.
AUCTION_ID 환경변수가 있으면 그 경매에 입찰하고, 없으면 active 목록의 첫 경매를 쓴다.
입찰은 422(너무 낮음)·429(rate limit)도 '정상 응답' 으로 센다 —
서버가 규칙대로 거절한 것이지 장애가 아니다.
"""

import os
import random
import uuid

from locust import HttpUser, between, task


class Bidder(HttpUser):
    wait_time = between(0.1, 0.5)  # 요청 사이 0.1~0.5초 (사람이 화면을 보는 시간)

    def on_start(self) -> None:
        email = f"load-{uuid.uuid4().hex[:8]}@example.com"
        self.client.post(
            "/auth/signup",
            json={"email": email, "password": "secret123", "nickname": "부하"},
            name="/auth/signup",
        )
        res = self.client.post(
            "/auth/login", json={"email": email, "password": "secret123"}, name="/auth/login"
        )
        self.headers = {"Authorization": f"Bearer {res.json()['access_token']}"}
        self.auction_id = os.environ.get("AUCTION_ID")
        if not self.auction_id:
            res = self.client.get("/auctions", params={"status": "active"}, name="/auctions")
            page = res.json()
            self.auction_id = page["items"][0]["id"] if page["items"] else None

    @task(5)
    def list_auctions(self) -> None:
        self.client.get("/auctions", params={"status": "active", "limit": 20}, name="/auctions")

    @task(3)
    def read_auction(self) -> None:
        if self.auction_id:
            self.client.get(f"/auctions/{self.auction_id}", name="/auctions/{id}")

    @task(1)
    def place_bid(self) -> None:
        if not self.auction_id:
            return
        # 대부분 '너무 낮음' 으로 거절된다 — 그래도 서버가 하는 일(락·규칙 검사)은 같다
        amount = random.randint(1, 500) * 1_000
        with self.client.post(
            f"/auctions/{self.auction_id}/bids",
            json={"amount": amount},
            headers=self.headers,
            name="/auctions/{id}/bids",
            catch_response=True,
        ) as res:
            if res.status_code in (201, 422, 429):
                res.success()
            else:
                res.failure(f"unexpected {res.status_code}")

FastAPI 10주 self-study 를 **실무 모양의 한 프로젝트**로 옮긴 캡스톤. 경매 등록 → 실시간 입찰 → 자동 마감 → 결제 웹훅까지,
